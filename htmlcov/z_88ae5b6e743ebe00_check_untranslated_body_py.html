1#!/usr/bin/env python3 

2"""Report post body lines that look untranslated (predominantly ASCII). 

3 

4Advisory check for `.github/workflows/post-quality.yml`. It replaces an inline 

5bash loop whose file selection was 

6``find _posts -name "*.md" -newer /tmp/quality-report.txt``. That report file is 

7written by an *earlier step of the same job*, after checkout has already written 

8every post, so it was always newer than all of them and the ``find`` returned 

9nothing. ``find`` exits 0 on no matches, so the ``||`` fallback never fired 

10either: the loop body never ran once. Real CI evidence (run 33591232799) — the 

11step printed its result 9.8 ms after starting, and reported 

12``Untranslated lines found: 0``. 

13 

14Selection here is by **filename date**, not mtime: deterministic, bounded, and 

15never silently empty. ``--min-files`` makes an empty scan an error instead of a 

16green no-op, because a check that inspects nothing is worse than no check — it 

17reports success. 

18 

19The ASCII threshold comes from ``common.summary_quality`` so this and 

20``check_description_quality.py`` cannot drift apart. 

21""" 

22 

23from __future__ import annotations 

24 

25import argparse 

26import re 

27import sys 

28from datetime import UTC, datetime, timedelta 

29from pathlib import Path 

30 

31_REPO_ROOT = Path(__file__).resolve().parent.parent.parent 

32if str(_REPO_ROOT / "scripts") not in sys.path: 

33 sys.path.insert(0, str(_REPO_ROOT / "scripts")) 

34 

35from common.summary_quality import ASCII_RATIO_THRESHOLD, ascii_ratio # noqa: E402 

36 

37_POST_DATE_RE = re.compile(r"^(\d{4}-\d{2}-\d{2})-") 

38_HANGUL_RE = re.compile(r"[가-힣]") 

39# Lines shorter than this are headings, table cells, and tickers — too short for 

40# a language judgement, and the same floor the report uses for descriptions. 

41_MIN_ALPHA = 30 

42 

43 

44def split_front_matter(text: str) -> tuple[str, str]: 

45 """Return ``(front_matter, body)``; the halves concatenate to *text*. 

46 

47 ``fix_untranslated_body.py`` rewrites only the body, so it needs the head 

48 back verbatim. Splitting in one place keeps the writer from disagreeing 

49 with the reader about where the body starts. 

50 """ 

51 if not text.startswith("---"): 

52 return "", text 

53 closing = text.find("\n---", 3) 

54 if closing == -1: 

55 return "", text 

56 return text[: closing + 4], text[closing + 4 :] 

57 

58 

59def split_body(text: str) -> str: 

60 """Return the post body, dropping YAML front matter.""" 

61 return split_front_matter(text)[1] 

62 

63 

64def is_untranslated(line: str) -> bool: 

65 """True if the line carries enough letters to judge and reads as English. 

66 

67 A line containing Hangul is Korean prose that quotes English entity names 

68 ("주요 공격 유형: Spot Price Manipulation(2건)", "Freight Technologies(FRGT) 

69 내부자가 …"), not an untranslated line. Judging those by ASCII ratio alone 

70 produced 196 findings on the 2026 corpus, mostly of that shape — and an 

71 advisory check that noisy gets ignored, which is the no-op it replaces. 

72 Untranslated means the translation never happened, so there is no Hangul. 

73 """ 

74 alpha = sum(1 for c in line if c.isalpha()) 

75 if alpha < _MIN_ALPHA: 

76 return False 

77 if _HANGUL_RE.search(line): 

78 return False 

79 return ascii_ratio(line) > ASCII_RATIO_THRESHOLD 

80

82def _is_scannable(line: str) -> bool: 

83 """Skip markup-only lines: HTML blocks, tables, links, and raw URLs carry 

84 ASCII by construction and say nothing about translation quality.""" 

85 stripped = line.strip() 

86 if not stripped or stripped.startswith(("<", "|", "#", "!", ">", "```")): 

87 return False 

88 return "http://" not in stripped and "https://" not in stripped

91def select_posts(posts_dir: Path, days: int) -> list[Path]: 

92 """Return posts whose *filename* date falls within the window.""" 

93 cutoff = (datetime.now(UTC) - timedelta(days=days)).date().isoformat() 

94 selected = [] 

95 for path in sorted(posts_dir.glob("*.md")): 

96 match = _POST_DATE_RE.match(path.name) 

97 if match and match.group(1) >= cutoff: 

98 selected.append(path) 

99 return selected 

100 

101 

102def scan_lines(body: str) -> list[str]: 

103 """Return the untranslated-looking lines of a post *body*. 

104 

105 The single detection entry point. ``fix_untranslated_body.py`` measures 

106 before/after with this same function so a repair pass cannot report success 

107 against a laxer rule than the check that reports the finding. 

108 """ 

109 return [line.strip() for line in body.splitlines() if _is_scannable(line) and is_untranslated(line)] 

110 

111 

112def scan(posts_dir: Path, days: int) -> tuple[int, list[tuple[Path, str]]]: 

113 """Return (files_scanned, findings).""" 

114 findings: list[tuple[Path, str]] = [] 

115 posts = select_posts(posts_dir, days) 

116 for path in posts: 

117 body = split_body(path.read_text(encoding="utf-8", errors="ignore")) 

118 for line in scan_lines(body): 

119 findings.append((path, line[:120])) 

120 return len(posts), findings 

121 

122 

123def main() -> int: 

124 parser = argparse.ArgumentParser(description="Detect untranslated English in post bodies") 

125 parser.add_argument("--days", type=int, default=7, help="Look back N days by filename date (default: 7)") 

126 parser.add_argument("--posts-dir", default=str(_REPO_ROOT / "_posts")) 

127 parser.add_argument( 

128 "--min-files", 

129 type=int, 

130 default=1, 

131 help="Fail if fewer than N posts were scanned — an empty scan must not report success (default: 1)", 

132 ) 

133 parser.add_argument( 

134 "--max-findings", 

135 type=int, 

136 default=-1, 

137 help="Exit 1 when findings exceed N. Negative means advisory only (default: -1)", 

138 ) 

139 args = parser.parse_args() 

140 

141 posts_dir = Path(args.posts_dir) 

142 if not posts_dir.is_dir(): 

143 print(f"posts dir not found: {posts_dir}", file=sys.stderr) 

144 return 2 

145 

146 scanned, findings = scan(posts_dir, args.days) 

147 print(f"Posts scanned (last {args.days}d by filename date): {scanned}") 

148 for path, line in findings: 

149 print(f"::warning file={path.as_posix()}::Untranslated: {line}") 

150 print(f"Untranslated lines found: {len(findings)}") 

151 

152 if scanned < args.min_files: 

153 print( 

154 f"::error::스캔한 포스트가 {scanned}건으로 --min-files={args.min_files} 미달 — " 

155 "검사가 실제로 아무것도 보지 않았다. 대상 선정이 깨졌는지 확인하라.", 

156 file=sys.stderr, 

157 ) 

158 return 1 

159 if args.max_findings >= 0 and len(findings) > args.max_findings: 

160 print(f"::error::번역 누락 의심 {len(findings)}건이 상한 {args.max_findings} 을 넘었다", file=sys.stderr) 

161 return 1 

162 return 0 

163 

164 

165if __name__ == "__main__": 

166 sys.exit(main())
