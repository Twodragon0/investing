266# Regex matching 3+ consecutive Latin-1 extended characters (U+00C0–U+00FF). 

267# A run of these in text that should be Korean/English strongly indicates 

268# mojibake (e.g. Latin-1 bytes re-decoded as UTF-8). 

269_MOJIBAKE_RE = re.compile(r"[\u00c0-\u00ff]{3,}") 

270 

271 

272def _is_mojibake(text: str) -> bool: 

273 """Return True if *text* looks like mojibake (corrupted encoding). 

274 

275 Heuristic: a sequence of 3 or more characters from the Latin-1 extended 

276 block (U+00C0–U+00FF) inside a string that is supposed to be Korean or 

277 English is a reliable indicator that Latin-1 bytes were misinterpreted as 

278 UTF-8. Legitimate Korean text never contains such runs. 

279 """ 

280 if not text: 

281 return False 

282 return bool(_MOJIBAKE_RE.search(text))

285def _cache_key(text: str) -> str: 

286 """SHA256 hash of text for cache key.""" 

287 return hashlib.sha256(text.encode("utf-8")).hexdigest()[:16]

290def _load_cache() -> Dict[str, str]: 

291 """Load translation cache from disk, cleaning artifact-contaminated entries.""" 

292 global _cache, _cache_dirty 

293 if _cache is not None: 

294 return _cache 

295 

296 _cache = {} 

297 if _CACHE_PATH.exists(): 

298 try: 

299 data = json.loads(_CACHE_PATH.read_text(encoding="utf-8")) 

300 if isinstance(data, dict): 

301 _cache = data 

302 except (json.JSONDecodeError, OSError) as e: 

303 logger.warning("Translation cache load failed: %s", e) 

304 _cache = {} 

305 

306 # Remove mojibake-corrupted entries before any further processing. 

307 mojibake_keys = [k for k, v in _cache.items() if _is_mojibake(v)] 

308 for k in mojibake_keys: 

309 del _cache[k] 

310 if mojibake_keys: 

311 _cache_dirty = True 

312 logger.warning( 

313 "Removed %d mojibake-corrupted entries from translation cache", 

314 len(mojibake_keys), 

315 ) 

316 

317 # Clean remaining entries with known translation artifacts. 

318 cleaned = 0 

319 keys_to_fix = [] 

320 for key, value in _cache.items(): 

321 fixed = _postprocess_translation(value) 

322 if fixed != value: 

323 keys_to_fix.append((key, fixed)) 

324 for key, fixed in keys_to_fix: 

325 _cache[key] = fixed 

326 cleaned += 1 

327 if cleaned: 

328 _cache_dirty = True 

329 logger.info("Cleaned %d cached translations with artifacts", cleaned) 

330 

331 logger.info("Translation cache loaded: %d entries", len(_cache)) 

332 return _cache

335def _save_cache() -> None: 

336 """Save translation cache atomically (temp file + rename).""" 

337 global _cache_dirty 

338 if not _cache_dirty or _cache is None: 

339 return 

340 

341 # Evict oldest entries if over limit (FIFO order) 

342 if len(_cache) > _MAX_CACHE_ENTRIES: 

343 evict_count = len(_cache) - _MAX_CACHE_ENTRIES 

344 keys = list(_cache.keys()) 

345 for k in keys[:evict_count]: 

346 del _cache[k] 

347 logger.info("Evicted %d old cache entries (limit: %d)", evict_count, _MAX_CACHE_ENTRIES) 

348 

349 # Strip any mojibake entries that may have been inserted this session. 

350 bad_keys = [k for k, v in _cache.items() if _is_mojibake(v)] 

351 for k in bad_keys: 

352 del _cache[k] 

353 if bad_keys: 

354 logger.warning( 

355 "Skipped saving %d mojibake-corrupted entries from translation cache", 

356 len(bad_keys), 

357 ) 

358 

359 _CACHE_PATH.parent.mkdir(parents=True, exist_ok=True) 

360 try: 

361 fd, tmp = tempfile.mkstemp(dir=str(_CACHE_PATH.parent), suffix=".tmp") 

362 with os.fdopen(fd, "w", encoding="utf-8") as f: 

363 json.dump(_cache, f, ensure_ascii=False) 

364 os.replace(tmp, str(_CACHE_PATH)) 

365 _cache_dirty = False 

366 except OSError as e: 

367 logger.warning("Translation cache save failed: %s", e)

375def _apply_term_overrides(text: str) -> tuple: 

376 """Replace known terms with placeholders before translation. 

377 

378 Returns (modified_text, replacements) where replacements is a list 

379 of (placeholder, korean_term) tuples for post-translation restoration. 

380 

381 Uses word boundaries (\\b) to prevent partial-word matches such as 

382 "AI" inside "gain" → "gAIn", or "SOL" inside "Gasoline" → "GaSOLine". 

383 """ 

384 replacements = [] 

385 result = text 

386 

387 # Sort by length (longest first) to avoid partial matches 

388 sorted_terms = sorted(_TERM_LOOKUP.items(), key=lambda x: len(x[0]), reverse=True) 

389 

390 for _lower_key, (original, korean) in sorted_terms: 

391 # Use word boundaries to prevent matching inside other words. 

392 # re.escape handles special chars (e.g. "S&P 500"); \b anchors to 

393 # word boundaries so "AI" won't match inside "gain" or "raise". 

394 escaped = re.escape(original) 

395 # Only wrap with \b when the term starts/ends with a word character 

396 left_b = r"\b" if re.match(r"\w", original[0]) else "" 

397 right_b = r"\b" if re.match(r"\w", original[-1]) else "" 

398 pattern = re.compile(left_b + escaped + right_b, re.IGNORECASE) 

399 if pattern.search(result): 

400 placeholder = f"__TERM{len(replacements)}__" 

401 result = pattern.sub(placeholder, result) 

402 replacements.append((placeholder, korean)) 

403 

404 return result, replacements

406 

407def _restore_terms(text: str, replacements: list) -> str: 

408 """Restore placeholders with Korean terms after translation.""" 

409 result = text 

410 for placeholder, korean in replacements: 

411 result = result.replace(placeholder, korean) 

412 return result

620def _postprocess_translation(text: str) -> str: 

621 """Fix common Google Translate artifacts in Korean translations. 

622 

623 Applied after term restoration, this function: 

624 1. Fixes token-name artifacts (PAIrs → Pairs) 

625 2. Corrects mistranslated media/source names 

626 3. Improves awkward Korean phrasing patterns 

627 """ 

628 

629 if not text: 

630 return text 

631 

632 # 1. Fix token artifacts (case-sensitive replacements) 

633 for wrong, correct in _TOKEN_ARTIFACT_MAP.items(): 

634 if wrong in text: 

635 text = text.replace(wrong, correct) 

636 

637 # 2. Fix mistranslated media names 

638 for wrong, correct in _MEDIA_NAME_FIXES.items(): 

639 if wrong in text: 

640 text = text.replace(wrong, correct) 

641 

642 # 3. Apply Korean style fixes (regex-based) 

643 for pattern, replacement in _KOREAN_STYLE_FIXES: 

644 text = re.sub(pattern, replacement, text) 

645 

646 return text.strip()

649def translate_to_korean(text: str) -> str: 

650 """Translate English text to Korean. Returns original on failure. 

651 

652 Uses Google Translate via deep-translator with term override 

653 protection and result caching. 

654 """ 

655 if not text or not text.strip(): 

656 return text 

657 

658 if not TRANSLATION_ENABLED: 

659 return text 

660 

661 # Check cache first 

662 cache = _load_cache() 

663 key = _cache_key(text) 

664 if key in cache: 

665 return cache[key] 

666 

667 try: 

668 from deep_translator import GoogleTranslator 

669 

670 # Protect known terms with placeholders 

671 modified, replacements = _apply_term_overrides(text) 

672 

673 translated = GoogleTranslator(source="en", target="ko").translate(modified) 

674 

675 if translated: 

676 # Restore Korean terms from placeholders 

677 result = _restore_terms(translated, replacements) 

678 # Post-process to fix Google Translate artifacts 

679 result = _postprocess_translation(result) 

680 

681 # Cache the result and flush to disk 

682 global _cache_dirty 

683 cache[key] = result 

684 _cache_dirty = True 

685 _save_cache() 

686 

687 return result 

688 except Exception as e: # noqa: BLE001 

689 logger.debug("Translation failed for '%s': %s", text[:50], e) 

690 

691 return text

694def translate_batch(texts: List[str]) -> List[str]: 

695 """Translate a list of English texts to Korean. 

696 

697 Processes in batches of 5 with 0.3s delay between batches 

698 to avoid rate limiting. Skips already-cached texts. 

699 """ 

700 if not texts or not TRANSLATION_ENABLED: 

701 return list(texts) 

702 

703 results = list(texts) # copy — fallback is original 

704 cache = _load_cache() 

705 

706 # Identify which texts need translation (not cached, not empty) 

707 to_translate: List[tuple] = [] # (index, text) 

708 for i, text in enumerate(texts): 

709 if not text or not text.strip(): 

710 continue 

711 key = _cache_key(text) 

712 if key in cache: 

713 results[i] = cache[key] 

714 else: 

715 to_translate.append((i, text)) 

716 

717 if not to_translate: 

718 return results 

719 

720 # Translate in batches 

721 for batch_start in range(0, len(to_translate), _BATCH_SIZE): 

722 batch = to_translate[batch_start : batch_start + _BATCH_SIZE] 

723 

724 if batch_start > 0: 

725 time.sleep(_BATCH_DELAY) 

726 

727 for idx, text in batch: 

728 translated = translate_to_korean(text) 

729 results[idx] = translated 

730 

731 _save_cache() 

732 return results

735_NON_TRANSLATABLE_LINE_RE = re.compile(r"^(?:#{1,6}\s|```|<[^>]+>|\|.*\||!\[.*\]\(|\[.*\]\(|https?://)") 

736_LEADING_MARKER_RE = re.compile(r"^(\s*(?:[-*+]\s+|\d+[.)]\s+|>\s+)?)") 

737_ENGLISH_SENTENCE_RE = re.compile(r"[A-Za-z][A-Za-z'’-]+")

740def _should_translate_body_line(line: str) -> bool: 

741 stripped = line.strip() 

742 if not stripped or _NON_TRANSLATABLE_LINE_RE.match(stripped): 

743 return False 

744 if "](" in stripped or "<picture" in stripped or "<img" in stripped: 

745 return False 

746 if re.search(r"[가-힣]", stripped): 

747 return False 

748 words = _ENGLISH_SENTENCE_RE.findall(stripped) 

749 return len(words) >= 4 and len(" ".join(words)) >= 24

752def translate_untranslated_body(content: str) -> str: 

753 """Translate obvious English prose lines in markdown body content. 

754 

755 Keeps markdown structure intact and only translates English-dominant lines 

756 that look like prose, while skipping links, HTML, tables, and image markup. 

757 """ 

758 if not content or not TRANSLATION_ENABLED: 

759 return content 

760 

761 lines = content.splitlines() 

762 targets: List[int] = [] 

763 payloads: List[str] = [] 

764 

765 for idx, line in enumerate(lines): 

766 if not _should_translate_body_line(line): 

767 continue 

768 marker_match = _LEADING_MARKER_RE.match(line) 

769 marker = marker_match.group(1) if marker_match else "" 

770 payload = line[len(marker) :].strip() 

771 if not payload: 

772 continue 

773 targets.append(idx) 

774 payloads.append(payload) 

775 

776 if not payloads: 

777 return content 

778 

779 translated = translate_batch(payloads) 

780 for idx, translated_line in zip(targets, translated, strict=False): 

781 marker_match = _LEADING_MARKER_RE.match(lines[idx]) 

782 marker = marker_match.group(1) if marker_match else "" 

783 lines[idx] = f"{marker}{translated_line.strip()}" 

784 

785 return "\n".join(lines)
