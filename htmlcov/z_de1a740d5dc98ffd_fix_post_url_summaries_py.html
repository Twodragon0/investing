1#!/usr/bin/env python3 

2"""Backfill per-URL summaries in published posts. 

3 

4``fix_post_descriptions.py`` repairs a post's front-matter ``description``. 

5This tool repairs the layer underneath it: the per-article blurbs rendered 

6inside the post body (``<p class="news-desc">`` cards and ``<span class="p0-desc">`` 

7alert entries). 

8 

9That layer was never measured. The front-matter report reads 99.3% real 

10content, while a 2026-08-06 scan over 2309 posts / 4504 cards found ~758 blurbs 

11carrying site chrome instead of article content — outlet self-introductions, 

12market-data error notices, navigation bars, newsletter solicitations. The 

13detector strengthening that keeps *new* posts clean ships separately; this 

14backfills what is already published. 

15 

16Sourcing order, mirroring the collection pipeline so a repaired blurb is 

17indistinguishable from a well-collected one: 

18 

191. **Re-fetch** the article URL (Google News redirects resolved first) and take 

20 its description, accepting it only if it passes the same quality gates a 

21 fresh collection would apply — not boilerplate, not a restatement of the 

22 title, long enough to inform. 

232. **Translate** to Korean when the recovered text is not already Korean. 

243. **Synthesize** from title + source when the fetch yields nothing usable 

25 (dead link, consent wall, paywall). 

26 

27Sampling 10 flagged URLs before this was written gave 7 usable re-fetches, so 

28the fetch path carries the bulk. 

29 

30**Run this incrementally.** 90% of flagged blurbs point at `news.google.com` 

31redirect links, and Google throttles the resolver hard: across three 

32back-to-back full passes the re-fetch yield fell 523 → 71 → 0, with the 

33resolver returning empty for every link by the third. Direct publisher URLs 

34kept working throughout, so the ceiling is the redirect resolver, not this 

35tool. Use `--limit` with a few hundred per run and space the runs out; a pass 

36that reports mostly "해결 실패" is a throttled pass, not a corpus without 

37recoverable summaries. 

38 

39Usage: 

40 python scripts/fix_post_url_summaries.py # dry-run report 

41 python scripts/fix_post_url_summaries.py --days 30 # recent posts only 

42 python scripts/fix_post_url_summaries.py --apply # write changes 

43 python scripts/fix_post_url_summaries.py --apply --limit 50 

44"""

80# A recovered blurb shorter than this says less than the headline already does. 

81_MIN_DESC_LEN = 30 

82 

83# …and longer than this is not a summary. `fetch_page_metadata` happily returns 

84# a whole article body when a page has no meta description; the first apply run 

85# put 400-800 character walls of text into cards sized for a sentence or two. 

86# Recovered text is trimmed back to a sentence boundary under this length. 

87_MAX_DESC_LEN = 300 

88 

89# Sentence terminators used to trim without cutting mid-word. 

90_SENTENCE_END_RE = re.compile(r"(?<=[.!?。])\s+|(?<=니다\.)\s*|(?<=습니다\.)\s*") 

91 

92# Card blurb: the anchor and its `<p class="news-desc">` sit in the same card 

93# div, with the source tag and severity badge in between. 

94_CARD_RE = re.compile( 

95 r'<a href="(?P<url>[^"]+)"[^>]*class="news-title"[^>]*>(?P<title>.*?)</a>' 

96 r'(?P<between>.*?)<p class="news-desc">(?P<desc>.*?)</p>', 

97 re.S, 

98) 

99 

100# Alert-box entry: `<a href=...>title</a> <span class="p0-desc">blurb</span>`. 

101_P0_RE = re.compile( 

102 r'<a href="(?P<url>[^"]+)"[^>]*>(?P<title>.*?)</a>\s*<span class="p0-desc">(?P<desc>.*?)</span>', 

103 re.S, 

104) 

105 

106_TAG_RE = re.compile(r"<[^>]+>") 

107_DATE_RE = re.compile(r"^(\d{4})-(\d{2})-(\d{2})-") 

108_HANGUL_RE = re.compile(r"[가-힣]")

111class Blurb(NamedTuple): 

112 """One per-URL summary found in a post body.""" 

113 

114 path: Path 

115 kind: str # "news-desc" | "p0-desc" 

116 url: str 

117 title: str 

118 raw: str # the exact inner text to replace, as it appears on disk 

119 text: str # unescaped, tag-stripped text for quality checks

122def _plain(fragment: str) -> str: 

123 """Tag-stripped, entity-decoded text for quality checks.""" 

124 return html.unescape(_TAG_RE.sub("", fragment)).strip()

127def _post_date(path: Path) -> date | None: 

128 match = _DATE_RE.match(path.name) 

129 if not match: 

130 return None 

131 try: 

132 return date(*(int(g) for g in match.groups())) 

133 except ValueError: 

134 return None

137def _is_bad(text: str, title: str) -> bool: 

138 """A blurb worth replacing: site chrome, or the headline restated.""" 

139 if not text: 

140 return True 

141 return is_boilerplate(text) or _is_desc_duplicate_of_title(text, title)

144def find_blurbs(path: Path) -> list[Blurb]: 

145 """Every per-URL summary in a post, flagged or not.""" 

146 text = path.read_text(encoding="utf-8", errors="replace") 

147 found: list[Blurb] = [] 

148 for kind, pattern in (("news-desc", _CARD_RE), ("p0-desc", _P0_RE)): 

149 for match in pattern.finditer(text): 

150 found.append( 

151 Blurb( 

152 path=path, 

153 kind=kind, 

154 url=match.group("url"), 

155 title=_plain(match.group("title")), 

156 raw=match.group("desc"), 

157 text=_plain(match.group("desc")), 

158 ) 

159 ) 

160 return found

163def collect_targets(posts_dir: Path, days: int | None) -> list[Blurb]: 

164 """Flagged blurbs across the corpus, newest post first.""" 

165 cutoff = None if days is None else datetime.now(UTC).date() - timedelta(days=days) 

166 targets: list[Blurb] = [] 

167 for path in sorted(posts_dir.glob("*.md"), reverse=True): 

168 posted = _post_date(path) 

169 if cutoff is not None and (posted is None or posted < cutoff): 

170 continue 

171 targets.extend(b for b in find_blurbs(path) if _is_bad(b.text, b.title)) 

172 return targets

175def _resolve(url: str) -> str: 

176 """Follow a Google News redirect to the publisher, best effort. 

177 

178 Host is parsed rather than substring-matched, so an article URL that merely 

179 mentions ``news.google.com`` in its path or query is not misrouted through 

180 the redirect resolver (CodeQL ``py/incomplete-url-substring-sanitization``). 

181 """ 

182 if not _is_google_news_host(url): 

183 return url 

184 try: 

185 resolved = _resolve_google_news_url(url) 

186 except Exception as exc: # network/parse failures are expected on old links 

187 logger.debug("Google News resolve failed for %s: %s", url[:60], exc) 

188 return "" 

189 return resolved if resolved and not _is_google_news_host(resolved) else ""

192def refetch(blurb: Blurb) -> str: 

193 """A usable replacement from the live article, or "" if none. 

194 

195 Applies the gates a fresh collection would: not boilerplate, not a 

196 restatement of the headline, related to the headline, long enough. 

197 """ 

198 link = _resolve(blurb.url) 

199 if not link: 

200 return "" 

201 try: 

202 meta = fetch_page_metadata(link, title=blurb.title) 

203 except Exception as exc: 

204 logger.debug("Fetch failed for %s: %s", link[:60], exc) 

205 return "" 

206 

207 # Unescape first: fetched text carries raw entities (`&hellip;`, `&amp;`). 

208 # Writing it back without this produced `&amp;hellip;` on the page — the 

209 # entity rendered as literal text instead of the character it names. 

210 desc = html.unescape((meta or {}).get("description", "")).strip() 

211 # Ad tails ride along with fetched copy ("Priority Gold에서 무료 가이드 받기"). 

212 # Delegated to the canonical stripper the quality checker already uses. 

213 desc = _trim_to_sentence(_strip_trailing_artifacts(desc)) 

214 if len(desc) < _MIN_DESC_LEN: 

215 return "" 

216 if is_boilerplate(desc) or _is_desc_duplicate_of_title(desc, blurb.title): 

217 return "" 

218 if not _is_title_related_description(blurb.title, desc): 

219 return "" 

220 

221 if not _HANGUL_RE.search(desc): 

222 try: 

223 translated = translate_to_korean(desc) 

224 except Exception as exc: 

225 logger.debug("Translation failed, keeping source text: %s", exc) 

226 translated = "" 

227 if translated and not is_boilerplate(translated): 

228 desc = translated 

229 

230 return desc

233def _trim_to_sentence(text: str, limit: int = _MAX_DESC_LEN) -> str: 

234 """Cut ``text`` back to the last sentence boundary at or under ``limit``. 

235 

236 Falls back to a hard cut with an ellipsis when the first sentence alone 

237 already exceeds the limit, so a run-on page never lands whole in a card. 

238 """ 

239 text = " ".join(text.split()) 

240 if len(text) <= limit: 

241 return text 

242 head = text[:limit] 

243 boundaries = [m.end() for m in _SENTENCE_END_RE.finditer(head)] 

244 if boundaries and boundaries[-1] >= _MIN_DESC_LEN: 

245 return head[: boundaries[-1]].strip() 

246 return head.rstrip() + "…"

249def synthesize(blurb: Blurb) -> str: 

250 """Fact-based fallback built from the headline when the link is unusable.""" 

251 try: 

252 return generate_synthetic_description(blurb.title, "", None).strip() 

253 except Exception as exc: 

254 logger.debug("Synthesis failed for %r: %s", blurb.title[:60], exc) 

255 return ""

258def replace_in_post(content: str, old_raw: str, new_text: str) -> tuple[str, bool]: 

259 """Swap one blurb's inner text, leaving the surrounding markup untouched. 

260 

261 The replacement is HTML-escaped because it lands inside an element body, 

262 and it is applied only when ``old_raw`` occurs exactly once — a blurb 

263 repeated verbatim in the same post would otherwise have every copy 

264 rewritten from one URL's fetch. 

265 """ 

266 if content.count(old_raw) != 1: 

267 return content, False 

268 escaped = html.escape(new_text, quote=False) 

269 return content.replace(old_raw, escaped, 1), True

272def is_google_news(blurb: Blurb) -> bool: 

273 """True when the blurb's link is a Google News redirect rather than an article.""" 

274 return _is_google_news_host(blurb.url)

277def _repair_one(blurb: Blurb, allow_synthesis: bool = True, direct_only: bool = False) -> tuple[Blurb, str, str]: 

278 """Resolve a replacement for one blurb. Returns (blurb, text, source).""" 

279 if direct_only and is_google_news(blurb): 

280 return blurb, "", "skipped" 

281 recovered = refetch(blurb) 

282 if recovered: 

283 return blurb, recovered, "refetch" 

284 if not allow_synthesis: 

285 return blurb, "", "unresolved" 

286 synthetic = synthesize(blurb) 

287 if synthetic and not _is_bad(synthetic, blurb.title): 

288 return blurb, synthetic, "synthetic" 

289 return blurb, "", "unresolved"

292def repair( 

293 targets: list[Blurb], 

294 workers: int, 

295 allow_synthesis: bool = True, 

296 direct_only: bool = False, 

297) -> list[tuple[Blurb, str, str]]: 

298 """Resolve replacements concurrently, preserving input order.""" 

299 results: list[tuple[Blurb, str, str] | None] = [None] * len(targets) 

300 with ThreadPoolExecutor(max_workers=workers) as pool: 

301 futures = {pool.submit(_repair_one, b, allow_synthesis, direct_only): i for i, b in enumerate(targets)} 

302 for future in as_completed(futures): 

303 index = futures[future] 

304 try: 

305 results[index] = future.result(timeout=60) 

306 except Exception as exc: 

307 logger.debug("Repair failed for %s: %s", targets[index].url[:60], exc) 

308 results[index] = (targets[index], "", "unresolved") 

309 return [r for r in results if r is not None]

312def apply_repairs(repairs: list[tuple[Blurb, str, str]]) -> tuple[int, int]: 

313 """Write resolved replacements to disk, grouped per post. 

314 

315 Returns ``(blurbs_written, posts_changed)``. Ambiguous anchors are skipped 

316 rather than guessed at. 

317 """ 

318 by_post: dict[Path, list[tuple[Blurb, str]]] = {} 

319 for blurb, text, _source in repairs: 

320 if text: 

321 by_post.setdefault(blurb.path, []).append((blurb, text)) 

322 

323 written = 0 

324 posts_written = 0 

325 for path, items in by_post.items(): 

326 content = path.read_text(encoding="utf-8", errors="replace") 

327 changed_here = 0 

328 for blurb, text in items: 

329 content, ok = replace_in_post(content, blurb.raw, text) 

330 if ok: 

331 changed_here += 1 

332 else: 

333 logger.warning("Skipped ambiguous blurb anchor in %s: %r", path.name, blurb.raw[:60]) 

334 if changed_here: 

335 path.write_text(content, encoding="utf-8") 

336 written += changed_here 

337 posts_written += 1 

338 return written, posts_written

341def format_report(repairs: list[tuple[Blurb, str, str]], applied: bool) -> str: 

342 """Human-readable summary with a sample of each outcome.""" 

343 total = len(repairs) 

344 counts = {"refetch": 0, "synthetic": 0, "unresolved": 0, "skipped": 0} 

345 for _blurb, _text, source in repairs: 

346 counts[source] = counts.get(source, 0) + 1 

347 

348 lines = [ 

349 f"URL 요약 백필 {'적용' if applied else '(dry-run)'}", 

350 f" 대상 블러브 : {total}", 

351 f" 재수집 성공 : {counts['refetch']}", 

352 f" 합성 대체 : {counts['synthetic']}", 

353 f" 해결 실패 : {counts['unresolved']}", 

354 f" 건너뜀 : {counts['skipped']}", 

355 ] 

356 

357 for source in ("refetch", "synthetic"): 

358 samples = [(b, t) for b, t, s in repairs if s == source][:3] 

359 if not samples: 

360 continue 

361 lines.append(f"\n --- {source} 샘플 ---") 

362 for blurb, text in samples: 

363 lines.append(f" [{blurb.path.name}] {blurb.title[:60]}") 

364 lines.append(f" before: {blurb.text[:80]}") 

365 lines.append(f" after : {text[:80]}") 

366 return "\n".join(lines)

369def main() -> int: 

370 parser = argparse.ArgumentParser(description="Backfill per-URL summaries in published posts.") 

371 parser.add_argument("--days", type=int, default=None, help="최근 N일 포스트만 (기본: 전체)") 

372 parser.add_argument("--limit", type=int, default=None, help="처리할 블러브 최대 개수") 

373 parser.add_argument("--workers", type=int, default=6, help="동시 재수집 스레드 수 (기본 6)") 

374 parser.add_argument("--apply", action="store_true", help="실제 파일에 기록 (기본: dry-run)") 

375 parser.add_argument( 

376 "--direct-only", 

377 action="store_true", 

378 help="Google News 리다이렉트는 건너뛰고 직접 퍼블리셔 URL 만 처리 (스로틀 중 유용)", 

379 ) 

380 parser.add_argument( 

381 "--skip-synthetic", 

382 action="store_true", 

383 help="재수집 실패 시 합성 폴백을 쓰지 않고 원문 유지 (품질 저하 방지)", 

384 ) 

385 args = parser.parse_args() 

386 

387 setup_logging() 

388 

389 targets = collect_targets(POSTS_DIR, args.days) 

390 if args.limit is not None: 

391 targets = targets[: args.limit] 

392 

393 if not targets: 

394 print("불량 URL 요약 없음.") 

395 return 0 

396 

397 logger.info("Repairing %d flagged blurbs with %d workers", len(targets), args.workers) 

398 repairs = repair( 

399 targets, 

400 args.workers, 

401 allow_synthesis=not args.skip_synthetic, 

402 direct_only=args.direct_only, 

403 ) 

404 

405 print(format_report(repairs, applied=args.apply)) 

406 

407 if args.apply: 

408 written, posts = apply_repairs(repairs) 

409 print(f"\n적용: 블러브 {written}건 / 포스트 {posts}개") 

410 else: 

411 print("\n(dry-run — 적용하려면 --apply)") 

412 return 0
