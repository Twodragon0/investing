141def load_entertainment_keywords(collector_name: str) -> frozenset: 

142 """collectors.yml에서 {collector_name}.keywords.entertainment_keywords를 로딩합니다. 

143 

144 로드 실패 또는 섹션 누락 시 _DEFAULT_ENTERTAINMENT_KEYWORDS로 fallback합니다. 

145 

146 Args: 

147 collector_name: collectors.yml 섹션 키 (예: "geopolitical", "crypto_news") 

148 Returns: 

149 frozenset — 소문자 엔터테인먼트 키워드 집합 

150 """ 

151 try: 

152 cfg = get_collector_config(collector_name) 

153 except Exception as exc: 

154 _log.debug("collectors.yml 로드 실패 (%s), 기본값 사용: %s", collector_name, exc) 

155 return _DEFAULT_ENTERTAINMENT_KEYWORDS 

156 

157 kw_cfg = cfg.get("keywords", {}) 

158 if not isinstance(kw_cfg, dict): 

159 _log.debug("collectors.yml: %s.keywords 섹션 없음, 기본값 사용", collector_name) 

160 return _DEFAULT_ENTERTAINMENT_KEYWORDS 

161 

162 ent_raw = kw_cfg.get("entertainment_keywords") 

163 if isinstance(ent_raw, list) and ent_raw: 

164 _log.debug("collectors.yml에서 %s.entertainment_keywords %d개 로드", collector_name, len(ent_raw)) 

165 return frozenset(ent_raw) 

166 

167 _log.debug("collectors.yml: %s.entertainment_keywords 누락 또는 빈 값, 기본값 사용", collector_name) 

168 return _DEFAULT_ENTERTAINMENT_KEYWORDS

171@lru_cache(maxsize=32) 

172def _keyword_pattern(keywords: frozenset) -> "re.Pattern[str]": 

173 """Build one word-boundary-anchored alternation for ``keywords``. 

174 

175 Bare substring matching (the previous ``any(kw in text ...)``) let short 

176 sports acronyms hit the middle of ordinary words. Measured on this repo's 

177 2026 corpus: ``nfl`` matched inflation/conflict/inflows (1,976 occurrences), 

178 ``nba`` matched coinbase/unbacked (853), ``hbo`` matched dashboard (214). 

179 Those are core crypto/macro vocabulary, so the filter was dropping the very 

180 items it exists to keep. 

181 

182 Every keyword is ASCII, so ``\\b`` is safe here. It would not be for Hangul: 

183 Python's ``\\w`` includes Hangul, so there is no boundary between two Korean 

184 characters — a Korean keyword would need lookarounds instead. 

185 

186 The boundary is applied per edge, only where the keyword's own edge is a word 

187 character. ``disney+`` ends in ``+``; a trailing ``\\b`` there would demand a 

188 word character right after the plus, so "disney+ streaming" would stop 

189 matching. Anchoring unconditionally silently narrows the filter. 

190 """ 

191 parts = [] 

192 for raw in sorted(keywords, key=len, reverse=True): 

193 kw = raw.strip() 

194 if not kw: 

195 continue 

196 # Explicit padding is the author's own anchor and is stricter than ``\b``. 

197 # ``" f1 "`` is padded precisely so it does not fire inside identifiers; 

198 # replacing that with ``\bf1\b`` makes it match the host in 

199 # ``https://f1.tokenpost.kr/...`` (``/`` and ``.`` are both boundaries). 

200 # Measured on the 2026 corpus: 70 hits, every one that CDN hostname. 

201 left = r"(?<!\S)" if raw[:1].isspace() else (r"\b" if kw[0].isalnum() or kw[0] == "_" else "") 

202 if raw[-1:].isspace(): 

203 right = r"(?!\S)" 

204 elif kw[-1].isalnum() or kw[-1] == "_": 

205 # Plural forms were matched for free by substring search ("oscar" 

206 # caught "Oscars"); a bare ``\b`` would drop them. An optional plural 

207 # suffix keeps those true positives without reopening the substring 

208 # hole, because the leading boundary still has to line up. 

209 right = r"(?:e?s)?\b" 

210 else: 

211 right = "" 

212 parts.append(f"{left}{re.escape(kw)}{right}") 

213 if not parts: 

214 return re.compile(r"(?!)") # never matches 

215 return re.compile("|".join(parts))

218def is_entertainment(item: Dict[str, Any], keywords: Optional[frozenset] = None) -> bool: 

219 """title + description에 엔터테인먼트/스포츠 키워드가 포함되면 True를 반환합니다. 

220 

221 키워드는 단어 경계에 고정해 비교합니다 — ``nfl`` 이 ``inflation`` 안에서 

222 매칭되던 오탐을 막습니다(``_keyword_pattern`` 주석의 실측 참조). 

223 

224 Args: 

225 item: title, description 키를 가진 뉴스 아이템 dict 

226 keywords: 사용할 키워드 집합. None이면 _DEFAULT_ENTERTAINMENT_KEYWORDS 사용 

227 Returns: 

228 True이면 필터 대상 (엔터테인먼트/스포츠 콘텐츠) 

229 """ 

230 kws = keywords if keywords is not None else _DEFAULT_ENTERTAINMENT_KEYWORDS 

231 text = (item.get("title", "") + " " + item.get("description", "")).lower() 

232 return bool(_keyword_pattern(frozenset(kws)).search(text))

235def filter_entertainment( 

236 items: List[Dict[str, Any]], 

237 keywords: Optional[frozenset] = None, 

238 logger: Optional[Any] = None, 

239) -> List[Dict[str, Any]]: 

240 """엔터테인먼트/스포츠 아이템을 목록에서 제거합니다. 

241 

242 Args: 

243 items: 뉴스 아이템 목록 

244 keywords: 사용할 키워드 집합. None이면 _DEFAULT_ENTERTAINMENT_KEYWORDS 사용 

245 logger: 필터링 건수를 기록할 logger. None이면 모듈 로거 사용 

246 Returns: 

247 필터링된 아이템 목록 

248 """ 

249 _logger = logger or _log 

250 filtered = [item for item in items if not is_entertainment(item, keywords)] 

251 removed = len(items) - len(filtered) 

252 if removed: 

253 _logger.debug("엔터테인먼트 필터: %d건 제거 (전체 %d건)", removed, len(items)) 

254 return filtered
