319class LoadRatio(NamedTuple): 

320 """파일럿 전·후의 분자/분모 원값. 비율은 메서드로 파생한다. 

321 

322 원값을 들고 다니는 이유: 후 구간 표본이 한 자릿수라 `0.67` 만 봐서는 2/3인지 

323 20/30인지 구분이 안 되는데 둘의 신뢰도는 전혀 다르다. 

324 """ 

325 

326 pre_num: int 

327 pre_den: int 

328 post_num: int 

329 post_den: int 

330 

331 def pre(self) -> float | None: 

332 """분모가 0이면 None — 0.0 을 돌려주면 '절감 100%' 로 읽힌다.""" 

333 return self.pre_num / self.pre_den if self.pre_den else None 

334 

335 def post(self) -> float | None: 

336 return self.post_num / self.post_den if self.post_den else None 

337 

338 def delta_pct(self) -> float | None: 

339 """전 대비 후의 변화율(%). 음수가 절감이다.""" 

340 pre, post = self.pre(), self.post() 

341 if pre is None or post is None or pre == 0: 

342 return None 

343 return (post - pre) / pre * 100 

344 

345 def delta_ci_pct(self, z: float = 1.96) -> tuple[float, float] | None: 

346 """변화율의 95% 신뢰구간(%). 네 칸 중 하나라도 0이면 None. 

347 

348 **이게 없으면 숫자가 실제보다 정밀해 보인다.** 2026-08-12 리뷰에서 드러났다: 

349 문서에 기록된 -33.7% 와 -33.3% 는 7시간 뒤 재측정에서 -23.3% 와 -10.7% 로 

350 갈렸는데, 구간을 붙여 보면 [-79%, +113%] 라 애초에 그 정도 흔들림이 예상 범위 

351 안이었다. "방향 참고용" 이라 부르려면 방향이 식별돼야 하는데 그렇지 않다. 

352 

353 네 칸 로그 비의 표준오차는 `sqrt(1/a + 1/b + 1/c + 1/d)` 다 (Katz). 로그 축에서 

354 대칭 구간을 만든 뒤 비율로 되돌린다 — 비율 축에서 직접 대칭 구간을 만들면 

355 하한이 음수가 되어 "절감 120%" 같은 값이 나온다. 

356 """ 

357 cells = (self.pre_num, self.pre_den, self.post_num, self.post_den) 

358 if any(c <= 0 for c in cells): 

359 return None 

360 se = math.sqrt(sum(1 / c for c in cells)) 

361 rr = (self.post_num / self.post_den) / (self.pre_num / self.pre_den) 

362 return ( 

363 (math.exp(math.log(rr) - z * se) - 1) * 100, 

364 (math.exp(math.log(rr) + z * se) - 1) * 100, 

365 )

370 denominator: list[datetime],

372 since: datetime | None = None,

463def filter_runs_after(runs: Sequence[Mapping[str, object]], pilot_start: datetime) -> list[Mapping[str, object]]: 

464 """파일럿 시작 **뒤에 시작한** 실행만 남긴다. 

465 

466 파일럿 전 실행은 코드가 없어 **구조적으로 skip 할 수 없다.** 분모에 넣으면 

467 skip 비율이 희석되는데, 그 방향이 나쁘다 — 2026-08-14 실측에서 조사한 20실행 중 

468 8건이 파일럿 전이라 5/20 = 25.0% (CI 상한 46.9%) 가 나왔다. 그 상한은 파일럿 전 

469 실측 no-op 비율 55.2% 를 **배제해서** "skip 이 기대보다 덜 걸린다" 는 잘못된 

470 인상을 준다. 후만 세면 5/12 = 41.7% CI [19.3%, 68.0%] 로 55.2% 를 정상적으로 

471 포함한다. 

472 

473 `createdAt` 을 못 읽는 실행은 **버린다.** 남기면 경계를 모르는 표본이 분모에 

474 들어가 같은 희석이 재발한다. 

475 """ 

476 kept: list[Mapping[str, object]] = [] 

477 for run in runs: 

478 started = run_started_at(run) 

479 if started is None: 

480 continue 

481 if started > pilot_start: 

482 kept.append(run) 

483 return kept

510def collect_skip_counts( 

511 workflow: str, limit: int, pilot_start: datetime 

512) -> tuple[int, int, str | None, list[list[str]]]: 

513 """gh CLI 로 워크플로우 실행 로그에서 skip 발생 횟수와 버린 파일 목록을 센다. 

514 

515 Returns (skip 횟수, 조사한 실행 수, 건너뛴 이유, skip 블록들). gh 가 없거나 

516 인증이 없으면 (0, 0, 이유, []) 로 graceful degradation. 

517 

518 `pilot_start` 뒤에 시작한 실행만 센다 — 이유는 `filter_runs_after` 참조. 

519 `--limit` 은 gh 에서 **가져올** 실행 수이고 필터 뒤 표본은 그보다 작다. 

520 

521 파일 목록까지 같은 로그에서 뽑는 이유: 오검출 판정([5])에 필요한데, 로그를 다시 

522 받으면 실행당 수 초가 두 배로 든다. 

523 

524 로그는 `LOG_FETCH_WORKERS` 개씩 동시에 받는다. 받아오지 못한 실행은 `checked` 

525 에서도 빠지므로 분모가 조용히 부풀지 않는다. 

526 """ 

527 try: 

528 listing = subprocess.run( 

529 [ 

530 "gh", 

531 "run", 

532 "list", 

533 f"--workflow={workflow}", 

534 f"--limit={limit}", 

535 "--json", 

536 "databaseId,conclusion,createdAt", 

537 ], 

538 cwd=REPO_ROOT, 

539 capture_output=True, 

540 text=True, 

541 check=True, 

542 ) 

543 except FileNotFoundError: 

544 return 0, 0, "gh CLI 미설치", [] 

545 except subprocess.CalledProcessError as e: 

546 return 0, 0, f"gh run list 실패 ({e.returncode})", [] 

547 

548 try: 

549 fetched = json.loads(listing.stdout) 

550 except json.JSONDecodeError: 

551 return 0, 0, "gh 출력 파싱 실패", [] 

552 

553 runs = filter_runs_after(fetched, pilot_start) 

554 if fetched and not count_runs_before(fetched, pilot_start): 

555 # 경계 앞 실행이 하나도 없다면 그 앞이 더 있는데 `--limit` 에서 잘렸을 수 

556 # 있다. 표본이 조용히 상한에 묶이는 것을 보이게 한다. 판정 기준이 "버려진 

557 # 실행 수" 가 아니라 "**경계 때문에** 버려진 실행 수" 인 이유: 시각을 못 읽어 

558 # 버린 실행 한 건이 이 안내를 꺼 버리면 안 된다. 

559 logger.info(" 가져온 %d실행에 파일럿 전이 없다 — --run-limit 을 올리면 표본이 늘 수 있다", len(fetched)) 

560 

561 run_ids: list[str] = [] 

562 for run in runs: 

563 run_id = str(run.get("databaseId", "")) 

564 if run_id: 

565 run_ids.append(run_id) 

566 

567 skips = 0 

568 checked = 0 

569 blocks: list[list[str]] = [] 

570 if run_ids: 

571 with ThreadPoolExecutor(max_workers=min(LOG_FETCH_WORKERS, len(run_ids))) as pool: 

572 # `map` 은 **입력 순서대로** 돌려준다. `as_completed` 로 바꾸면 blocks 순서가 

573 # 로그 도착 순서를 타서 같은 데이터에서 실행마다 다른 출력이 나온다 — 

574 # 오검출 목록([5])이 흔들리면 재현 대조가 불가능해진다. 

575 for log in pool.map(fetch_run_log, run_ids): 

576 if log is None: 

577 continue 

578 checked += 1 

579 found = parse_skip_paths(log) 

580 if found: 

581 skips += 1 

582 blocks.extend(found) 

583 if checked != len(run_ids): 

584 # 분모가 시도한 수보다 작다는 사실 자체를 보이게 한다. 위 warning 은 실패한 

585 # 실행마다 나오지만, 합계가 없으면 몇 건이 빠졌는지 눈으로 세야 한다. 

586 logger.warning(" 로그 %d/%d 만 수집됐다 — skip 비율의 분모가 그만큼 작다", checked, len(run_ids)) 

587 return skips, checked, None, blocks

701def wilson_interval(successes: int, total: int, z: float = 1.96) -> tuple[float, float] | None: 

702 """비율의 Wilson 95% 신뢰구간. 표본이 0이면 None. 

703 

704 정규근사(Wald)를 쓰지 않는 이유: 후 구간 표본이 한 자릿수이고 skip 비율이 0 이나 

705 1 에 가까울 때 Wald 구간은 [0,1] 을 벗어나거나 폭이 0으로 붕괴한다. 설계 문서의 

706 CI 폭 표도 Wilson 이라 같은 척도를 유지한다. 

707 """ 

708 if total <= 0: 

709 return None 

710 p = successes / total 

711 denom = 1 + z**2 / total 

712 center = (p + z**2 / (2 * total)) / denom 

713 margin = z * math.sqrt(p * (1 - p) / total + z**2 / (4 * total**2)) / denom 

714 return max(0.0, center - margin), min(1.0, center + margin)

736def classify_commit(paths: list[str], whitelist: frozenset[str]) -> str:
