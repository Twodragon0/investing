416def parse_shard(spec: str) -> tuple[int, int]: 

417 """``"2/5"`` -> ``(2, 5)``. Index is 1-based, as it appears in CI job names.""" 

418 try: 

419 index_text, total_text = spec.split("/", 1) 

420 index, total = int(index_text), int(total_text) 

421 except ValueError as exc: 

422 raise SystemExit(f"--shard 형식은 'N/M' 이다 (받은 값: {spec!r})") from exc 

423 if total < 1 or not (1 <= index <= total): 

424 raise SystemExit(f"--shard 범위 오류: 1 <= N <= M 이어야 한다 (받은 값: {spec!r})") 

425 return index, total

428def select_shard(items: list, shard: tuple[int, int] | None) -> list: 

429 """Round-robin slice of ``items`` for one shard. 

430 

431 Round-robin rather than contiguous blocks: cases are grouped by guard file 

432 in source order, so a contiguous split would put every coverage case in one 

433 shard and every tree-write case in another, making shard runtimes lopsided. 

434 Interleaving spreads the slow ones evenly. 

435 

436 Every item lands in exactly one shard for any (index, total), so a run split 

437 across shards covers the same set as an unsharded run — asserted in 

438 ``tests/test_guard_falsifiability.py``. 

439 """ 

440 if shard is None: 

441 return list(items) 

442 index, total = shard 

443 return [item for position, item in enumerate(items) if position % total == index - 1]

446def apply_static_mutation(source: str, case: StaticCase) -> str: 

447 """정적 케이스의 변형을 적용한다. 

448 

449 앵커가 0회 또는 2회 이상이면 조용히 첫 일치를 바꾸는 대신 예외를 던진다 — 

450 엉뚱한 줄을 바꾼 mutation 은 가드를 통과시켜 VACUOUS 오탐을 만든다. 

451 """ 

452 if case.old is None: 

453 return source + case.new 

454 occurrences = source.count(case.old) 

455 if occurrences != 1: 

456 raise RuntimeError( 

457 f"{case.target}: 앵커가 정확히 1회가 아니다 (found={occurrences}). " 

458 f"엉뚱한 위치를 변형하면 VACUOUS 오탐이 난다. 앵커: {case.old!r}" 

459 ) 

460 return source.replace(case.old, case.new, 1)

468def _purge_pycache() -> None: 

469 """__pycache__ 전량 삭제. 

470 

471 conftest.py 를 짧은 간격으로 반복 덮어쓰면 pyc 검증(mtime+size)이 변경을 

472 놓쳐 스테일 바이트코드가 재사용된다. 초기 하네스가 이 때문에 실행마다 

473 다른 결과를 냈다. 

474 """ 

475 for path in (REPO_ROOT / "tests", REPO_ROOT / "scripts"): 

476 for cache_dir in path.rglob("__pycache__"): 

477 shutil.rmtree(cache_dir, ignore_errors=True)

480def _disable_fixture(src: str, name: str) -> str: 

481 """지정 fixture 의 autouse=True 를 False 로 바꾼다. 

482 

483 데코레이터에 다른 kwarg(``scope="session"`` 등)가 함께 있어도 매칭한다 — 

484 ``autouse=True`` 만 치환하고 나머지 인자는 보존한다. 

485 """ 

486 pattern = re.compile(r"@pytest\.fixture\(([^)]*)autouse=True([^)]*)\)\n(def " + re.escape(name) + r"\()") 

487 patched, count = pattern.subn(r"@pytest.fixture(\1autouse=False\2)\n\3", src) 

488 if count != 1: 

489 raise RuntimeError(f"fixture {name!r} 의 autouse 데코레이터를 정확히 1개 찾지 못했다 (found={count})") 

490 return patched

528def _assert_safe_to_run() -> None: 

529 """하네스가 건드릴 파일에 커밋되지 않은 변경이 있으면 중단한다. 

530 

531 하네스는 대상 파일을 덮어썼다 복원한다. 미커밋 변경이 있는 상태에서 중간에

535 proc = subprocess.run( 

536 ["git", "status", "--porcelain", "--", *targets], 

537 cwd=REPO_ROOT, 

538 capture_output=True, 

539 text=True, 

540 check=False, 

541 ) 

542 if proc.stdout.strip(): 

543 raise SystemExit( 

544 "중단: 하네스가 변형할 파일에 커밋되지 않은 변경이 있다. 덮어썼다 " 

545 "복원하는 방식이므로 먼저 커밋하거나 stash 할 것.\n" + proc.stdout

549def _snapshot_state() -> dict[Path, bytes]: 

550 """`_state/` 파일 내용을 스냅샷한다. 

551 

552 `module-level:image_rejection_metrics` 케이스는 import 시점 리다이렉트를 

553 일부러 깨뜨리므로, 그 실행의 atexit flush 가 진짜 

554 `_state/image_rejection_metrics.json` 에 기록된다 — 리다이렉트가 막던 바로 

555 그 오염이다. 하네스가 남긴 이 부작용은 하네스가 되돌려야 한다. 

556 """ 

557 state_dir = REPO_ROOT / "_state" 

558 if not state_dir.is_dir(): 

559 return {} 

560 return {p: p.read_bytes() for p in state_dir.rglob("*") if p.is_file()}

563def _restore_state(snapshot: dict[Path, bytes]) -> list[str]: 

564 """스냅샷과 달라진 `_state/` 파일을 되돌리고 복원한 목록을 돌려준다.""" 

565 restored = [] 

566 for path, content in snapshot.items(): 

567 if not path.exists() or path.read_bytes() != content: 

568 path.parent.mkdir(parents=True, exist_ok=True) 

569 path.write_bytes(content) 

570 restored.append(str(path.relative_to(REPO_ROOT))) 

571 return restored

574def _verdict(rc_mutated: int, rc_control: int) -> str: 

575 if rc_mutated != 0 and rc_control == 0: 

576 return "FALSIFIABLE" 

577 if rc_mutated == 0: 

578 return "VACUOUS" 

579 return "CONTROL-FAIL"

582def _run_static_cases(shard: tuple[int, int] | None = None) -> list[dict]: 

583 """정적 가드 케이스를 검증한다 (각각 자기 대상 파일만 변형/복원).""" 

584 results: list[dict] = [] 

585 for case in select_shard(list(STATIC_CASES), shard): 

586 target = REPO_ROOT / case.target 

587 original = target.read_text(encoding="utf-8") 

588 try: 

589 target.write_text(apply_static_mutation(original, case), encoding="utf-8") 

590 rc_mutated = _run_node(case.node_id) 

591 target.write_text(original, encoding="utf-8") 

592 rc_control = _run_node(case.node_id) 

593 except RuntimeError as exc: # 앵커 불일치 — 변형 자체가 무효 

594 results.append( 

595 { 

596 "fixture": case.label, 

597 "guard": case.node_id, 

598 "patched_rc": None, 

599 "control_rc": None, 

600 "verdict": "AMBIGUOUS-ANCHOR", 

601 "detail": str(exc), 

602 } 

603 ) 

604 continue 

605 finally: 

606 target.write_text(original, encoding="utf-8") 

607 _purge_pycache() 

608 

609 results.append( 

610 { 

611 "fixture": case.label, 

612 "guard": case.node_id, 

613 "patched_rc": rc_mutated, 

614 "control_rc": rc_control, 

615 "verdict": _verdict(rc_mutated, rc_control), 

616 } 

617 ) 

618 return results

621def run_all(shard: tuple[int, int] | None = None) -> list[dict]: 

622 """모든 케이스를 검증하고 결과 리스트를 돌려준다. 

623 

624 ``shard`` 가 주어지면 해당 샤드에 배정된 케이스만 실행한다. 드리프트 검사 

625 (CASES 미등록 fixture)는 샤드와 무관하게 항상 전수로 돈다 — 특정 샤드에서만 

626 보이는 미등록 fixture 는 없고, 누락은 어느 샤드에서든 즉시 드러나야 한다. 

627 """ 

628 _assert_safe_to_run() 

629 original = CONFTEST.read_text(encoding="utf-8") 

630 state_snapshot = _snapshot_state() 

631 

632 unmapped = [f for f in discover_autouse_fixtures(original) if f not in CASES] 

633 fixture_cases = select_shard(list(CASES.items()), shard) 

634 

635 results: list[dict] = [] 

636 try: 

637 for name, node in fixture_cases: 

638 patched = ( 

639 _disable_module_level(original) if name == _MODULE_LEVEL_CASE else _disable_fixture(original, name) 

640 ) 

641 

642 CONFTEST.write_text(patched, encoding="utf-8") 

643 rc_patched = _run_guard(node) 

644 

645 CONFTEST.write_text(original, encoding="utf-8") 

646 rc_control = _run_guard(node) 

647 

648 results.append( 

649 { 

650 "fixture": name, 

651 "guard": node, 

652 "patched_rc": rc_patched, 

653 "control_rc": rc_control, 

654 "verdict": _verdict(rc_patched, rc_control), 

655 } 

656 ) 

657 finally: 

658 CONFTEST.write_text(original, encoding="utf-8") 

659 restored = _restore_state(state_snapshot) 

660 if restored: 

661 print(f"[guard-falsifiability] _state 복원: {', '.join(restored)}", file=sys.stderr) 

662 _purge_pycache() 

663 

664 for fixture in unmapped: 

665 results.append( 

666 { 

667 "fixture": fixture, 

668 "guard": None, 

669 "patched_rc": None, 

670 "control_rc": None, 

671 "verdict": "UNMAPPED", 

672 } 

673 ) 

674 

675 results.extend(_run_static_cases(shard)) 

676 return results

679def _render_table(results: list[dict]) -> str: 

680 lines = [] 

681 for r in results: 

682 rc = f"patched_rc={r['patched_rc']} control_rc={r['control_rc']}" 

683 lines.append(f"{r['verdict']:17} | {rc:32} | {r['fixture']:44} | {r['guard'] or '(가드 없음)'}") 

684 ok = sum(1 for r in results if r["verdict"] == "FALSIFIABLE") 

685 lines.append("") 

686 lines.append(f"{ok}/{len(results)} guards falsifiable") 

687 for r in results: 

688 if r["verdict"] != "FALSIFIABLE": 

689 lines.append(f" {r['verdict']}: {r['fixture']} -> {r['guard'] or 'CASES 미등록'}") 

690 return "\n".join(lines)

693def main() -> int: 

694 parser = argparse.ArgumentParser(description="격리 가드 falsifiability 검증") 

695 mode = parser.add_mutually_exclusive_group() 

696 mode.add_argument("--json", action="store_true", help="JSON 출력") 

697 mode.add_argument("--check", action="store_true", help="vacuous/미등록 시 exit 1 (CI)") 

698 parser.add_argument("--shard", metavar="N/M", help="N번째/M개 샤드만 실행 (CI 매트릭스용)") 

699 args = parser.parse_args() 

700 

701 results = run_all(parse_shard(args.shard) if args.shard else None) 

702 

703 if args.json: 

704 print(json.dumps(results, ensure_ascii=False, indent=2)) 

705 else: 

706 print(_render_table(results)) 

707 

708 failed = [r for r in results if r["verdict"] != "FALSIFIABLE"] 

709 if args.check and failed: 

710 return 1 

711 return 0
