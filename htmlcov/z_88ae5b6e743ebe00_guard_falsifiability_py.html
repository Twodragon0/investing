1205def parse_shard(spec: str) -> tuple[int, int]: 

1206 """``"2/5"`` -> ``(2, 5)``. Index is 1-based, as it appears in CI job names.""" 

1207 try: 

1208 index_text, total_text = spec.split("/", 1) 

1209 index, total = int(index_text), int(total_text) 

1210 except ValueError as exc: 

1211 raise SystemExit(f"--shard 형식은 'N/M' 이다 (받은 값: {spec!r})") from exc 

1212 if total < 1 or not (1 <= index <= total): 

1213 raise SystemExit(f"--shard 범위 오류: 1 <= N <= M 이어야 한다 (받은 값: {spec!r})") 

1214 return index, total

1217def select_shard(items: list, shard: tuple[int, int] | None) -> list: 

1218 """Round-robin slice of ``items`` for one shard. 

1219 

1220 Round-robin rather than contiguous blocks: cases are grouped by guard file 

1221 in source order, so a contiguous split would put every coverage case in one 

1222 shard and every tree-write case in another, making shard runtimes lopsided. 

1223 Interleaving spreads the slow ones evenly. 

1224 

1225 Every item lands in exactly one shard for any (index, total), so a run split 

1226 across shards covers the same set as an unsharded run — asserted in 

1227 ``tests/test_guard_falsifiability.py``. 

1228 """ 

1229 if shard is None: 

1230 return list(items) 

1231 index, total = shard 

1232 return [item for position, item in enumerate(items) if position % total == index - 1]
