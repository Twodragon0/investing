598def parse_shard(spec: str) -> tuple[int, int]: 

599 """``"2/5"`` -> ``(2, 5)``. Index is 1-based, as it appears in CI job names.""" 

600 try: 

601 index_text, total_text = spec.split("/", 1) 

602 index, total = int(index_text), int(total_text) 

603 except ValueError as exc: 

604 raise SystemExit(f"--shard 형식은 'N/M' 이다 (받은 값: {spec!r})") from exc 

605 if total < 1 or not (1 <= index <= total): 

606 raise SystemExit(f"--shard 범위 오류: 1 <= N <= M 이어야 한다 (받은 값: {spec!r})") 

607 return index, total

610def select_shard(items: list, shard: tuple[int, int] | None) -> list: 

611 """Round-robin slice of ``items`` for one shard. 

612 

613 Round-robin rather than contiguous blocks: cases are grouped by guard file 

614 in source order, so a contiguous split would put every coverage case in one 

615 shard and every tree-write case in another, making shard runtimes lopsided. 

616 Interleaving spreads the slow ones evenly. 

617 

618 Every item lands in exactly one shard for any (index, total), so a run split 

619 across shards covers the same set as an unsharded run — asserted in 

620 ``tests/test_guard_falsifiability.py``. 

621 """ 

622 if shard is None: 

623 return list(items) 

624 index, total = shard 

625 return [item for position, item in enumerate(items) if position % total == index - 1]

628def apply_static_mutation(source: str, case: StaticCase) -> str: 

629 """정적 케이스의 변형을 적용한다. 

630 

631 앵커가 0회 또는 2회 이상이면 조용히 첫 일치를 바꾸는 대신 예외를 던진다 — 

632 엉뚱한 줄을 바꾼 mutation 은 가드를 통과시켜 VACUOUS 오탐을 만든다. 

633 """ 

634 if case.old is None: 

635 return source + case.new 

636 occurrences = source.count(case.old) 

637 if occurrences != 1: 

638 raise RuntimeError( 

639 f"{case.target}: 앵커가 정확히 1회가 아니다 (found={occurrences}). " 

640 f"엉뚱한 위치를 변형하면 VACUOUS 오탐이 난다. 앵커: {case.old!r}" 

641 ) 

642 return source.replace(case.old, case.new, 1)

653 워크플로우가 `on.pull_request.paths` 에 같은 목록을 손으로 유지하면 반드시 

654 드리프트한다 — 새 STATIC_CASES 가 새 파일을 겨냥해도 트리거는 모르므로, 

655 그 가드는 falsifiability 검증 없이 머지된다. 그래서 목록을 케이스 정의에서 

656 **파생**시키고, 워크플로우는 `--list-targets` 로 이걸 읽는다. 

657 

658 포함 대상: 

659 

660 * 하네스가 덮어썼다 복원하는 파일(`_mutated_files`) — 변형 대상이 바뀌면 

661 앵커가 어긋날 수 있다; 

662 * 각 케이스가 돌리는 가드 테스트 파일(node id 의 파일 부분) — 가드 본체가 

663 바뀌면 여전히 falsifiable 한지 다시 봐야 한다; 

664 * 하네스 자신과 그 워크플로우. 

665 """ 

666 paths = {str(p.relative_to(REPO_ROOT)) for p in _mutated_files()} 

667 paths.update(case.node_id.split("::", 1)[0] for case in STATIC_CASES) 

668 paths.add(str(Path(__file__).resolve().relative_to(REPO_ROOT))) 

669 paths.add(".github/workflows/guard-falsifiability.yml") 

670 return sorted(paths)

673def _purge_pycache() -> None: 

674 """__pycache__ 전량 삭제. 

675 

676 conftest.py 를 짧은 간격으로 반복 덮어쓰면 pyc 검증(mtime+size)이 변경을 

677 놓쳐 스테일 바이트코드가 재사용된다. 초기 하네스가 이 때문에 실행마다 

678 다른 결과를 냈다. 

679 """ 

680 for path in (REPO_ROOT / "tests", REPO_ROOT / "scripts"): 

681 for cache_dir in path.rglob("__pycache__"): 

682 shutil.rmtree(cache_dir, ignore_errors=True)

685def _disable_fixture(src: str, name: str) -> str: 

686 """지정 fixture 의 autouse=True 를 False 로 바꾼다. 

687 

688 데코레이터에 다른 kwarg(``scope="session"`` 등)가 함께 있어도 매칭한다 — 

689 ``autouse=True`` 만 치환하고 나머지 인자는 보존한다. 

690 """ 

691 pattern = re.compile(r"@pytest\.fixture\(([^)]*)autouse=True([^)]*)\)\n(def " + re.escape(name) + r"\()") 

692 patched, count = pattern.subn(r"@pytest.fixture(\1autouse=False\2)\n\3", src) 

693 if count != 1: 

694 raise RuntimeError(f"fixture {name!r} 의 autouse 데코레이터를 정확히 1개 찾지 못했다 (found={count})") 

695 return patched

698def _disable_module_level(src: str) -> str: 

699 """import 시점 리다이렉트가 ImportError 를 맞도록 모듈명을 비존재로 바꾼다.""" 

700 if _MODULE_LEVEL_IMPORT not in src: 

701 raise RuntimeError("conftest 에서 모듈 레벨 image_rejection_metrics import 를 찾지 못했다") 

702 return src.replace( 

703 _MODULE_LEVEL_IMPORT, 

704 " import common.image_rejection_metrics_ABSENT as _irm_a", 

705 1, 

706 )

709def _run_node(node_id: str) -> int: 

710 """pytest node id 하나만 실행하고 종료 코드를 돌려준다.""" 

711 _purge_pycache() 

712 env = {**os.environ, "PYTHONDONTWRITEBYTECODE": "1"} 

713 proc = subprocess.run( 

714 [sys.executable, "-B", "-m", "pytest", node_id, "-q", "--no-cov", "-p", "no:randomly"], 

715 cwd=REPO_ROOT, 

716 capture_output=True, 

717 text=True, 

718 env=env, 

719 ) 

720 return proc.returncode

733def _assert_safe_to_run() -> None: 

734 """하네스가 건드릴 파일에 커밋되지 않은 변경이 있으면 중단한다. 

735 

736 하네스는 대상 파일을 덮어썼다 복원한다. 미커밋 변경이 있는 상태에서 중간에 

737 죽으면 사용자의 작업이 사라질 수 있다. 

738 """ 

739 targets = sorted({str(p) for p in _mutated_files()}) 

740 proc = subprocess.run( 

741 ["git", "status", "--porcelain", "--", *targets], 

742 cwd=REPO_ROOT, 

743 capture_output=True, 

744 text=True, 

745 check=False, 

746 ) 

747 if proc.stdout.strip(): 

748 raise SystemExit( 

749 "중단: 하네스가 변형할 파일에 커밋되지 않은 변경이 있다. 덮어썼다 " 

750 "복원하는 방식이므로 먼저 커밋하거나 stash 할 것.\n" + proc.stdout 

751 )

754def _snapshot_state() -> dict[Path, bytes]: 

755 """`_state/` 파일 내용을 스냅샷한다. 

756 

757 `module-level:image_rejection_metrics` 케이스는 import 시점 리다이렉트를 

758 일부러 깨뜨리므로, 그 실행의 atexit flush 가 진짜 

759 `_state/image_rejection_metrics.json` 에 기록된다 — 리다이렉트가 막던 바로 

760 그 오염이다. 하네스가 남긴 이 부작용은 하네스가 되돌려야 한다. 

761 """ 

762 state_dir = REPO_ROOT / "_state" 

763 if not state_dir.is_dir(): 

764 return {} 

765 return {p: p.read_bytes() for p in state_dir.rglob("*") if p.is_file()}

768def _restore_state(snapshot: dict[Path, bytes]) -> list[str]: 

769 """스냅샷과 달라진 `_state/` 파일을 되돌리고 복원한 목록을 돌려준다.""" 

770 restored = [] 

771 for path, content in snapshot.items(): 

772 if not path.exists() or path.read_bytes() != content: 

773 path.parent.mkdir(parents=True, exist_ok=True) 

774 path.write_bytes(content) 

775 restored.append(str(path.relative_to(REPO_ROOT))) 

776 return restored

779def _verdict(rc_mutated: int, rc_control: int) -> str: 

780 if rc_mutated != 0 and rc_control == 0: 

781 return "FALSIFIABLE" 

782 if rc_mutated == 0: 

783 return "VACUOUS" 

784 return "CONTROL-FAIL"

787def _run_static_cases(shard: tuple[int, int] | None = None) -> list[dict]: 

788 """정적 가드 케이스를 검증한다 (각각 자기 대상 파일만 변형/복원).""" 

789 results: list[dict] = [] 

790 for case in select_shard(list(STATIC_CASES), shard): 

791 target = REPO_ROOT / case.target 

792 original = target.read_text(encoding="utf-8") 

793 try: 

794 target.write_text(apply_static_mutation(original, case), encoding="utf-8") 

795 rc_mutated = _run_node(case.node_id) 

796 target.write_text(original, encoding="utf-8") 

797 rc_control = _run_node(case.node_id) 

798 except RuntimeError as exc: # 앵커 불일치 — 변형 자체가 무효 

799 results.append( 

800 { 

801 "fixture": case.label, 

802 "guard": case.node_id, 

803 "patched_rc": None, 

804 "control_rc": None, 

805 "verdict": "AMBIGUOUS-ANCHOR", 

806 "detail": str(exc), 

807 } 

808 ) 

809 continue 

810 finally: 

811 target.write_text(original, encoding="utf-8") 

812 _purge_pycache() 

813 

814 results.append( 

815 { 

816 "fixture": case.label, 

817 "guard": case.node_id, 

818 "patched_rc": rc_mutated, 

819 "control_rc": rc_control, 

820 "verdict": _verdict(rc_mutated, rc_control), 

821 } 

822 ) 

823 return results

826def run_all(shard: tuple[int, int] | None = None) -> list[dict]: 

827 """모든 케이스를 검증하고 결과 리스트를 돌려준다. 

828 

829 ``shard`` 가 주어지면 해당 샤드에 배정된 케이스만 실행한다. 드리프트 검사 

830 (CASES 미등록 fixture)는 샤드와 무관하게 항상 전수로 돈다 — 특정 샤드에서만 

831 보이는 미등록 fixture 는 없고, 누락은 어느 샤드에서든 즉시 드러나야 한다. 

832 """ 

833 _assert_safe_to_run() 

834 original = CONFTEST.read_text(encoding="utf-8") 

835 state_snapshot = _snapshot_state() 

836 

837 unmapped = [f for f in discover_autouse_fixtures(original) if f not in CASES] 

838 fixture_cases = select_shard(list(CASES.items()), shard) 

839 

840 results: list[dict] = [] 

841 try: 

842 for name, node in fixture_cases: 

843 patched = ( 

844 _disable_module_level(original) if name == _MODULE_LEVEL_CASE else _disable_fixture(original, name) 

845 ) 

846 

847 CONFTEST.write_text(patched, encoding="utf-8") 

848 rc_patched = _run_guard(node) 

849 

850 CONFTEST.write_text(original, encoding="utf-8") 

851 rc_control = _run_guard(node) 

852 

853 results.append( 

854 { 

855 "fixture": name, 

856 "guard": node, 

857 "patched_rc": rc_patched, 

858 "control_rc": rc_control, 

859 "verdict": _verdict(rc_patched, rc_control), 

860 } 

861 ) 

862 finally: 

863 CONFTEST.write_text(original, encoding="utf-8") 

864 restored = _restore_state(state_snapshot) 

865 if restored: 

866 print(f"[guard-falsifiability] _state 복원: {', '.join(restored)}", file=sys.stderr) 

867 _purge_pycache() 

868 

869 for fixture in unmapped: 

870 results.append( 

871 { 

872 "fixture": fixture, 

873 "guard": None, 

874 "patched_rc": None, 

875 "control_rc": None, 

876 "verdict": "UNMAPPED", 

877 } 

878 ) 

879 

880 results.extend(_run_static_cases(shard)) 

881 return results

884def _render_table(results: list[dict]) -> str: 

885 lines = [] 

886 for r in results: 

887 rc = f"patched_rc={r['patched_rc']} control_rc={r['control_rc']}" 

888 lines.append(f"{r['verdict']:17} | {rc:32} | {r['fixture']:44} | {r['guard'] or '(가드 없음)'}") 

889 ok = sum(1 for r in results if r["verdict"] == "FALSIFIABLE") 

890 lines.append("") 

891 lines.append(f"{ok}/{len(results)} guards falsifiable") 

892 for r in results: 

893 if r["verdict"] != "FALSIFIABLE": 

894 lines.append(f" {r['verdict']}: {r['fixture']} -> {r['guard'] or 'CASES 미등록'}") 

895 return "\n".join(lines)

898def main() -> int: 

899 parser = argparse.ArgumentParser(description="격리 가드 falsifiability 검증") 

900 mode = parser.add_mutually_exclusive_group() 

901 mode.add_argument("--json", action="store_true", help="JSON 출력") 

902 mode.add_argument("--check", action="store_true", help="vacuous/미등록 시 exit 1 (CI)") 

903 mode.add_argument( 

904 "--list-targets", 

905 action="store_true", 

906 help="변경 시 이 하네스를 돌려야 하는 저장소-상대 경로를 한 줄씩 출력 (워크플로우 트리거 판정용)", 

907 ) 

908 parser.add_argument("--shard", metavar="N/M", help="N번째/M개 샤드만 실행 (CI 매트릭스용)") 

909 args = parser.parse_args() 

910 

911 if args.list_targets: 

912 print("\n".join(trigger_paths())) 

913 return 0 

914 

915 results = run_all(parse_shard(args.shard) if args.shard else None) 

916 

917 if args.json: 

918 print(json.dumps(results, ensure_ascii=False, indent=2)) 

919 else: 

920 print(_render_table(results)) 

921 

922 failed = [r for r in results if r["verdict"] != "FALSIFIABLE"] 

923 if args.check and failed: 

924 return 1 

925 return 0

929 sys.exit(main())
