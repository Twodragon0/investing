124def _run(cmd: Sequence[str]) -> str | None: 

125 """외부 명령을 돌린다. 실패하면 None — 호출부가 집계를 중단한다. 

126

128 레코드 0건이면 모든 커밋이 거절로 보고된다. 

129 """ 

130 try: 

131 out = subprocess.run( 

132 list(cmd), 

133 cwd=REPO_ROOT, 

134 capture_output=True, 

135 text=True, 

136 check=True, 

137 ) 

138 except FileNotFoundError: 

139 logger.error("%s 를 찾을 수 없다", cmd[0]) 

140 return None 

141 except subprocess.CalledProcessError as e: 

142 logger.error("%s 실패 (rc %d): %s", " ".join(cmd), e.returncode, (e.stderr or "")[:300]) 

143 return None 

144 return out.stdout

147def parse_records(payload: str, env: str) -> tuple[list[Record], int | None]: 

148 """`vercel ls -F json` 한 페이지를 파싱한다. 

149 

150 Returns (레코드들, 다음 페이지 커서). 커서가 None 이면 더 없다. 

151 """ 

152 try: 

153 data = json.loads(payload) 

154 except json.JSONDecodeError: 

155 logger.error("vercel 출력 파싱 실패 (%s)", env) 

156 return [], None 

157 

158 records: list[Record] = [] 

159 for item in data.get("deployments") or []: 

160 created = item.get("createdAt") 

161 if not isinstance(created, (int, float)): 

162 continue 

163 meta = item.get("meta") or {} 

164 records.append( 

165 Record( 

166 at=datetime.fromtimestamp(created / 1000, KST), 

167 env=env, 

168 state=str(item.get("state") or ""), 

169 sha=meta.get("githubCommitSha"), 

170 subject=str(meta.get("githubCommitMessage") or ""), 

171 ) 

172 ) 

173 cursor = (data.get("pagination") or {}).get("next") 

174 return records, cursor if isinstance(cursor, int) else None

190def fetch_records(project: str, env: str, since: datetime) -> list[Record] | None: 

191 """`--next` 로 `since` 를 덮을 때까지 페이지네이션한다. 실패하면 None. 

192 

193 페이지 상한에 걸리면 **None 이다.** 부분 결과를 돌려주면 덜 덮인 구간의 커밋이 

194 전부 "레코드 없음 = 거절" 로 보고되는데, 그게 종료 코드 0 으로 나가면 조용히 

195 틀린 집계가 성공으로 읽힌다. 모듈 앞머리의 `_run` 폴백 원칙과 같은 이유다. 

196 """ 

197 collected: list[Record] = [] 

198 cursor: int | None = None 

199 for _ in range(MAX_PAGES): 

200 cmd = vercel_list_cmd(project, env) 

201 if cursor is not None: 

202 cmd += ["--next", str(cursor)] 

203 payload = _run(cmd) 

204 if payload is None: 

205 return None 

206 page, next_cursor = parse_records(payload, env) 

207 if not page: 

208 break 

209 collected += page 

210 oldest = min(r.at for r in page) 

211 logger.debug(" %s +%d건 (누적 %d) oldest=%s", env, len(page), len(collected), oldest) 

212 if oldest < since or next_cursor is None: 

213 break 

214 if next_cursor == cursor: 

215 # 커서가 안 움직이면 같은 페이지를 상한까지 다시 쌓는다 — 레코드가 

216 # 중복 계상되어 피크가 부풀려진다. 진전이 없으면 멈춘다. 

217 logger.warning(" %s 커서 정체 (%s) — 페이지네이션을 멈춘다", env, next_cursor) 

218 break 

219 cursor = next_cursor 

220 else: 

221 logger.error(" %s 페이지 상한 %d 도달 — 구간이 덜 덮였다", env, MAX_PAGES) 

222 return None 

223 return collected

226def fetch_commits(since: datetime) -> list[Commit] | None: 

227 """`origin/main` 커밋을 시각 오름차순으로 가져온다. 실패하면 None. 

228 

229 `git log` 는 최신 먼저이고, **커밋 시각이 같으면 자식을 부모보다 먼저** 낸다 

230 (그래프 순회 순서). 그대로 안정 정렬하면 동률 구간에서 head 가 non-head 보다 

231 앞에 남는다. `classify_commits` 는 뒤만 보므로 그 non-head 를 거절로 오분류한다 

232 — 이 도구가 없애려던 바로 그 오류가 되돌아온다. 

233 

234 그래서 정렬 **전에 뒤집는다.** 뒤집으면 동률 구간이 부모→자식 순이 되고, 

235 안정 정렬이 그 순서를 보존한다. 

236 """ 

237 raw = _run(["git", "log", "origin/main", f"--since={since.isoformat()}", "--format=%H|%cI|%s"]) 

238 if raw is None: 

239 return None 

240 commits: list[Commit] = [] 

241 for line in raw.splitlines(): 

242 parts = line.split("|", 2) 

243 if len(parts) != 3: 

244 continue 

245 sha, stamp, subject = parts 

246 try: 

247 at = datetime.fromisoformat(stamp).astimezone(KST) 

248 except ValueError: 

249 continue 

250 commits.append(Commit(at=at, sha=sha, subject=subject)) 

251 commits.reverse() 

252 commits.sort(key=lambda c: c.at) 

253 return commits

256def rolling_peak( 

257 times: Sequence[datetime], 

258 window: timedelta, 

259 *, 

260 min_end: datetime | None = None, 

261 max_end: datetime | None = None, 

262) -> tuple[datetime, int] | None: 

263 """창을 미끄러뜨리며 센 최대 레코드 수와 그 창의 종료 시각. 

264 

265 창은 `(t - window, t]` — 오른쪽 닫힘이다. 경계 규칙이 다르면 피크가 1건 

266 달라진다(재측정 문서의 94 와 이 도구의 95 가 그 차이다). 

267 

268 `min_end`/`max_end` 는 **창 종료 시각**을 제한할 뿐 세는 레코드를 거르지 

269 않는다. 파일럿 전/후를 나눌 때 이 구분이 중요하다 — 레코드 쪽을 걸러 버리면 

270 경계를 걸친 창에서 실제 쿼터 부하를 과소평가한다. 쿼터 카운터는 파일럿이 

271 언제 머지됐는지 모른다. 반대로 창 종료만 제한하면 경계를 걸친 창이 "파일럿 후" 

272 로 잘못 귀속되는데, 호출부가 `min_end = 파일럿 + window` 로 창 전체를 후 

273 구간에 넣어 그 문제를 피한다. 

274 """ 

275 if not times: 

276 return None 

277 ordered = sorted(times) 

278 best: tuple[datetime, int] | None = None 

279 left = 0 

280 for right, end in enumerate(ordered): 

281 while ordered[left] <= end - window: 

282 left += 1 

283 if min_end is not None and end < min_end: 

284 continue 

285 if max_end is not None and end > max_end: 

286 break 

287 count = right - left + 1 

288 if best is None or count > best[1]: 

289 best = (end, count) 

290 return best

293def classify_commits( 

294 commits: Sequence[Commit], 

295 deployed_shas: frozenset[str], 

296 batch_window: timedelta, 

297) -> Verdict: 

298 """레코드를 받은 커밋 / 푸시 batch non-head / 거절 로 가른다. 

299 

300 non-head 판정은 **뒤를 본다.** `batch_window` 안의 후속 커밋 중 하나라도 

301 레코드를 받았으면 이 커밋은 같은 푸시의 head 가 아니었다는 뜻이다. 

302 """ 

303 deployed: list[Commit] = [] 

304 non_head: list[Commit] = [] 

305 rejected: list[Commit] = [] 

306 for i, commit in enumerate(commits): 

307 if commit.sha in deployed_shas: 

308 deployed.append(commit) 

309 continue 

310 batched = False 

311 for follower in commits[i + 1 :]: 

312 if follower.at - commit.at > batch_window: 

313 break 

314 if follower.sha in deployed_shas: 

315 batched = True 

316 break 

317 (non_head if batched else rejected).append(commit) 

318 return Verdict(deployed=deployed, non_head=non_head, rejected=rejected)

321def record_kind(record: Record) -> str: 

322 """레코드를 축으로 가른다. preview 는 환경으로, 나머지는 커밋 제목으로 판정한다. 

323

325 `[1]`·`[2]` 의 대상 집합을 정하므로 "구성 표시용" 이 아니다. 

326 """

356def filter_kind(records: Sequence[Record], kind: str) -> list[Record]: 

357 """`--kind` 로 레코드를 거른다. 

358 

359 **거른 피크는 쿼터 헤드룸이 아니다.** Vercel 의 쿼터 카운터는 우리 분류를 모르고 

360 통에 든 레코드를 전부 센다. `--kind collector` 로 나온 36 은 "쿼터를 36 만큼 

361 쓰고 있다" 가 아니라 "그 창의 부하 중 36 이 수집기 축이다" 다. 둘을 섞으면 

362 임계(추정 하한 ~95)와 비교해 여유가 있다고 오독한다. 

363 

364 이 렌즈가 필요한 이유는 반대 방향의 오독을 막기 위해서다. 2026-08-14 실측에서 

365 24h 피크가 95 → 52 로 떨어졌는데 그 창의 수집기 레코드는 30 → 36 으로 **늘었다.** 

366 총량만 보면 파일럿 성과로 읽히지만 빠진 43건은 전부 개발 활동(preview 43→11, 

367 PR 머지 16→4)이었다. 파일럿이 움직일 수 있는 축만 따로 세야 그 착시가 안 생긴다. 

368 """ 

369 keep = KIND_FILTERS[kind] 

370 if keep is None: 

371 return list(records) 

372 return [r for r in records if keep(record_kind(r))]

376 """`(end - window, end]` 창에 든 레코드를 종류별로 센다."""

380def _report_composition(records: Sequence[Record], end: datetime, window: timedelta, label: str) -> None: 

381 counts = window_composition(records, end, window) 

382 total = sum(counts.values()) 

383 if not total: 

384 return 

385 logger.info(" %s 창 구성 (총 %d건)", label, total) 

386 for kind, count in counts.most_common(): 

387 logger.info(" %-12s %3d (%d%%)", kind, count, round(count / total * 100)) 

388 dev = total - counts.get("수집기", 0) 

389 logger.info(" → 개발 활동 소계 %d (%d%%)", dev, round(dev / total * 100))

392def report_peaks( 

393 records: Sequence[Record], 

394 window: timedelta, 

395 pilot_merged: datetime | None, 

396 since: datetime, 

397 kind: str = DEFAULT_KIND, 

398) -> None: 

399 """롤링 피크와 그 창의 구성을 낸다. 

400 

401 `records` 는 `since` **이전 것까지** 받는다. 페이지네이션이 오버슈트해서 

402 가져온 그 레코드들을 버리면, `since` 직후에 끝나는 창이 자기 앞부분을 잃어 

403 피크가 과소평가된다 — 쿼터 카운터는 집계 시작일을 모른다. 대신 `min_end` 로 

404 **창 종료**만 `since` 이후로 제한한다. 

405 

406 `records` 는 **거르지 않은 전량**이고 `--kind` 필터는 이 함수 안에서 건다. 

407 호출부가 거른 목록을 만들어 넘기는 구조로 두면 그 목록이 `main()` 의 지역변수로 

408 남고, 같은 타입의 전량 목록과 뒤바뀔 수 있다. 뒤바뀌면 조용히 틀린다 — 특히 

409 `[3]` SHA 대조가 거른 목록을 집으면 걸러진 레코드를 받은 커밋이 전부 "레코드 

410 없음 = 거절" 이 되어 거절 수가 통째로 조작된다. 필터를 안으로 넣으면 그 뒤바뀜이 

411 **표현 불가능**해진다. 

412 

413 구성 표는 언제나 전량으로 낸다 — 거른 뒤 구성을 내면 "수집기 100%" 라는 자명한 

414 줄만 남아, 그 창의 실제 쿼터 부하가 얼마였는지가 출력에서 사라진다. 그 숫자가 

415 이 렌즈를 쓰는 이유의 절반이다. 

416 """ 

417 hours = int(window.total_seconds() // 3600) 

418 composition = records 

419 records = filter_kind(records, kind) 

420 logger.info("[1] %dh 롤링 피크 (%s)", hours, KIND_LABELS[kind]) 

421 if kind != DEFAULT_KIND: 

422 logger.info(" ※ 축을 걸렀다 — 아래 피크는 **쿼터 부하가 아니다.** 구성 표의 총계가 부하다.") 

423 

424 times = [r.at for r in records] 

425 overall = rolling_peak(times, window, min_end=since) 

426 if overall is None: 

427 logger.warning(" 레코드 0건 — 피크를 낼 수 없다") 

428 return 

429 at, count = overall 

430 logger.info(" 전 기간 피크 %d건 (창 종료 %s KST)", count, at.strftime("%Y-%m-%d %H:%M")) 

431 

432 if pilot_merged is None: 

433 _report_composition(composition, at, window, "피크") 

434 return 

435 

436 # 창 **전체**가 한쪽 구간에 들어가는 것만 본다. 경계를 걸친 창은 전/후 어느 

437 # 쪽 부하도 아니라서 어디에 넣어도 그 구간을 오해하게 만든다. 

438 times = [r.at for r in records] 

439 for label, bounds in ( 

440 ("파일럿 전", {"max_end": pilot_merged}), 

441 ("파일럿 후", {"min_end": pilot_merged + window}), 

442 ): 

443 peak = rolling_peak(times, window, **bounds) 

444 if peak is None: 

445 logger.info(" %s 피크 창 없음 (구간이 %dh 보다 짧다)", label, hours) 

446 continue 

447 peak_at, peak_count = peak 

448 logger.info(" %s 피크 %d건 (창 종료 %s)", label, peak_count, peak_at.strftime("%m-%d %H:%M")) 

449 _report_composition(composition, peak_at, window, label) 

450 

451 logger.info(" 주의: 피크는 파일럿이 건드리지 않는 축(개발 활동)이 지배할 수 있다.") 

452 logger.info(" 총량 하락을 파일럿 성과로 읽기 전에 위 창 구성을 볼 것.") 

453 if kind == DEFAULT_KIND: 

454 logger.info(" `--kind collector` 로 파일럿이 움직일 수 있는 축만 따로 볼 수 있다.") 

455

457def report_daily(records: Sequence[Record], window: timedelta, since: datetime, kind: str = DEFAULT_KIND) -> None: 

458 """일자별 레코드 수와 그날의 롤링 최대. 

459 

460 `report_peaks` 와 같은 이유로 `since` 이전 레코드도 **세는 데는** 쓴다. 표시만 

461 `since` 이후 날짜로 자른다 — 안 그러면 첫날의 롤링 최대가 그날 레코드 수와 

462 같아지는 가짜 계단이 생긴다. 

463 

464 `report_peaks` 와 같이 전량을 받아 **안에서** 거른다(그 docstring 참조). 

465 """ 

466 records = filter_kind(records, kind) 

467 logger.info("[2] 일자별 (KST) — 레코드 수 / 그날의 롤링 최대 (%s)", KIND_LABELS[kind]) 

468 if kind != DEFAULT_KIND: 

469 # `[1]` 과 `[2]` 는 따로 인용된다. 경고를 `[1]` 에만 두면 `[2]` 표만 떼어 

470 # 읽는 사람에게는 이 숫자가 쿼터 부하로 보인다. 

471 logger.info(" ※ 축을 걸렀다 — 쿼터 부하가 아니다.") 

472 ordered = sorted(records, key=lambda r: r.at) 

473 times = [r.at for r in ordered] 

474 per_day: Counter[str] = Counter(r.at.strftime("%Y-%m-%d") for r in ordered if r.at >= since) 

475 day_peak: dict[str, int] = {} 

476 left = 0 

477 for right, end in enumerate(times): 

478 while times[left] <= end - window: 

479 left += 1 

480 if end < since: 

481 continue 

482 day = end.strftime("%Y-%m-%d") 

483 day_peak[day] = max(day_peak.get(day, 0), right - left + 1) 

484 for day in sorted(per_day): 

485 logger.info(" %s 레코드 %3d 롤링 피크 %3d", day, per_day[day], day_peak.get(day, 0)) 

486

488def load_at(times: Sequence[datetime], at: datetime, window: timedelta) -> int: 

489 """`(at - window, at]` 에 든 레코드 수 = 그 순간의 쿼터 부하. 

490 

491 `rolling_peak` 과 **같은 경계 규칙**(오른쪽 닫힘)이다. 규칙이 갈리면 같은 데이터에서 

492 피크와 부하가 1건씩 어긋난다. 

493 

494 `rolling_peak` 과 다른 점은 평가 격자다. `rolling_peak` 은 **레코드** 시각에서 

495 창을 끝내고 여기는 **커밋** 시각에서 끝낸다. 두 격자는 일치하지 않으므로 이 함수의 

496 최대값이 `rolling_peak` 의 피크보다 작을 수 있다 — 틀린 게 아니라 다른 질문이다 

497 ("부하의 최대" vs "이 커밋이 겪은 부하"). 

498 """ 

499 return sum(1 for t in times if at - window < t <= at)

502def _percentiles(values: Sequence[int]) -> str: 

503 """분포 요약 한 줄. 거절 부하를 비교할 기준선이 없으면 숫자를 읽을 수 없다.""" 

504 ordered = sorted(values) 

505 n = len(ordered) 

506 return ( 

507 f"n={n} min={ordered[0]} p25={ordered[n // 4]} " 

508 f"중앙값={ordered[n // 2]} p75={ordered[3 * n // 4]} max={ordered[-1]}" 

509 )

512def report_rejections( 

513 verdict: Verdict, 

514 pilot_merged: datetime | None, 

515 records: Sequence[Record] | None = None, 

516 window: timedelta | None = None, 

517) -> None: 

518 """SHA 대조 결과를 낸다. 

519 

520 `records`·`window` 를 주면 거절마다 **그 순간의 롤링 부하**를 붙이고 정상 생성의 

521 부하 분포를 함께 낸다. 이게 없으면 "거절 25건" 만 남아 쿼터 때문인 거절과 그렇지 

522 않은 거절이 구분되지 않는다 — `branch-protection.md` 의 "쿼터로 설명되지 않는 

523 단발 거절" 절이 손으로 계산해야 했던 숫자다. 손계산은 이 파일에서 두 번 틀렸다 

524 (91→79, 94→95). 도구가 내면 다시 틀리지 않는다. 

525 """ 

526 logger.info("[3] SHA 대조 — main 커밋이 production 레코드를 받았는가") 

527 total = len(verdict.deployed) + len(verdict.non_head) + len(verdict.rejected) 

528 logger.info(" main 커밋 %d건", total) 

529 logger.info(" 레코드 생성 %d", len(verdict.deployed)) 

530 logger.info(" 푸시 batch non-head %d", len(verdict.non_head)) 

531 logger.info(" 레코드 없음 = 거절 %d", len(verdict.rejected)) 

532 if not verdict.rejected: 

533 return 

534 

535 # sha → 그 커밋이 겪은 롤링 부하. `records`/`window` 가 없으면 None 이고 그때는 

536 # 부하 열을 아예 내지 않는다 — 0 으로 폴백하면 "부하 0에서 거절" 로 읽힌다. 

537 loads: dict[str, int] | None = None 

538 if records and window is not None: 

539 times = [r.at for r in records] 

540 loads = {c.sha: load_at(times, c.at, window) for c in (*verdict.deployed, *verdict.rejected)} 

541 if verdict.deployed: 

542 logger.info( 

543 " 정상 생성 %d건의 %dh 롤링 부하 분포: %s", 

544 len(verdict.deployed), 

545 int(window.total_seconds() // 3600), 

546 _percentiles([loads[c.sha] for c in verdict.deployed]), 

547 ) 

548 

549 by_day = Counter(c.at.strftime("%Y-%m-%d") for c in verdict.rejected) 

550 logger.info(" 일자별 거절: %s", ", ".join(f"{d} {n}" for d, n in sorted(by_day.items()))) 

551 for commit in verdict.rejected: 

552 if loads is None: 

553 logger.info(" %s %s", commit.at.strftime("%m-%d %H:%M"), commit.subject[:60]) 

554 else: 

555 logger.info( 

556 " %s 부하 %3d %s", 

557 commit.at.strftime("%m-%d %H:%M"), 

558 loads[commit.sha], 

559 commit.subject[:60], 

560 ) 

561 if pilot_merged is not None: 

562 after = sum(1 for c in verdict.rejected if c.at > pilot_merged) 

563 logger.info(" 파일럿 후 거절 %d건", after) 

564 logger.info(" 주의: 포화 사건이 드물어 거절 0건은 성공의 증거가 아니다.") 

565

568 parser = argparse.ArgumentParser(description=__doc__) 

569 parser.add_argument("--project", default=DEFAULT_PROJECT, help=f"Vercel 프로젝트 (기본: {DEFAULT_PROJECT})")

577 parser.add_argument("--pilot-merged", help="파일럿 머지 시각 ISO — 주면 전/후를 나눠 낸다") 

578 parser.add_argument( 

579 "--batch-window-seconds", 

580 type=int, 

581 default=DEFAULT_BATCH_SECONDS, 

582 help=f"푸시 batch non-head 판정 간격 (기본: {DEFAULT_BATCH_SECONDS})", 

583 ) 

584 parser.add_argument( 

585 "--kind", 

586 choices=sorted(KIND_FILTERS), 

587 default=DEFAULT_KIND, 

588 help=( 

589 f"롤링 피크·일자별을 낼 축 (기본: {DEFAULT_KIND}). " 

590 "collector 는 수집기 커밋만, dev 는 그 외(preview·PR 머지 등)만 센다. " 

591 "거른 피크는 쿼터 부하가 아니라 부하의 구성이다 — [3] SHA 대조에는 적용되지 않는다." 

592 ), 

593 ) 

594 args = parser.parse_args() 

595 

596 if args.window_hours <= 0: 

597 logger.error("--window-hours 는 양수여야 한다: %d", args.window_hours) 

598 return 2 

599 

600 now = datetime.now(KST) 

601 if args.since: 

602 try: 

603 since = datetime.fromisoformat(args.since) 

604 except ValueError: 

605 logger.error("--since 파싱 실패: %s", args.since) 

606 return 2 

607 if since.tzinfo is None: 

608 since = since.replace(tzinfo=KST) 

609 else: 

610 since = now - timedelta(days=14) 

611 

612 pilot_merged: datetime | None = None 

613 if args.pilot_merged: 

614 try: 

615 pilot_merged = datetime.fromisoformat(args.pilot_merged) 

616 except ValueError: 

617 logger.error("--pilot-merged 파싱 실패: %s", args.pilot_merged) 

618 return 2 

619 if pilot_merged.tzinfo is None: 

620 logger.error("--pilot-merged 에 타임존이 없다: %s", args.pilot_merged) 

621 return 2 

622 

623 window = timedelta(hours=args.window_hours) 

624 logger.info("Vercel 쿼터 집계 — %s 이후, %dh 창", since.strftime("%Y-%m-%d %H:%M"), args.window_hours) 

625 

626 records: list[Record] = [] 

627 for env in ENVIRONMENTS: 

628 page = fetch_records(args.project, env, since) 

629 if page is None: 

630 logger.error("레코드 수집 실패 — 집계를 중단한다 (부분 집계는 거절을 부풀린다)") 

631 return 2 

632 records += page 

633 if not records: 

634 logger.error("레코드 0건 — vercel 프로젝트 이름과 인증을 확인할 것") 

635 return 2

638 # 가져온 `since` 이전 레코드까지 넘긴다 (`report_peaks` docstring 참조). 

639 in_window = [r for r in records if r.at >= since] 

640 counts = Counter(r.env for r in in_window) 

641 logger.info( 

642 " 수집 %d건 (production %d / preview %d), 창 계산용 선행 레코드 %d건",

644 counts.get("production", 0), 

645 counts.get("preview", 0), 

646 len(records) - len(in_window), 

647 ) 

648 

649 # `--kind` 는 **보고 축만** 좁힌다. 여기서 거른 목록을 만들지 않는 것이 요점이다 — 

650 # 지역변수로 남으면 아래 `deployed_shas` 가 그것을 집을 수 있고, 그러면 걸러진 

651 # 레코드를 받은 커밋이 전부 "레코드 없음 = 거절" 이 되어 거절 수가 통째로 조작된다. 

652 # 필터는 두 report 함수 **안**에 있다. 

653 report_peaks(records, window, pilot_merged, since, args.kind) 

654 report_daily(records, window, since, args.kind) 

655 

656 commits = fetch_commits(since) 

657 if commits is None: 

658 logger.error("git log 실패 — SHA 대조를 건너뛴다") 

659 return 2 

660 deployed_shas = frozenset(r.sha for r in records if r.env == "production" and r.sha) 

661 verdict = classify_commits(commits, deployed_shas, timedelta(seconds=args.batch_window_seconds)) 

662 report_rejections(verdict, pilot_merged, records, window) 

663 

664 return 0
