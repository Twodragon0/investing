374def _extract_raw_patterns(title: str) -> tuple: 

375 """Run regex extractions on title and return (tickers, values, proper, kr_entities).""" 

376 tickers = re.findall(r"\b[A-Z]{2,5}\b", title) 

377 values = re.findall(r"\$[\d,.]+[KkMmBbTt]?|\d+(?:\.\d+)?%", title) 

378 kr_entities = re.findall(r"[가-힣]{2,}", title) 

379 proper = re.findall(r"\b[A-Z][a-z]{2,}\b", title) 

380 return tickers, values, proper, kr_entities 

381 

382 

383def _filter_entities(tickers: list, proper: list) -> tuple: 

384 """Filter noise from ticker and proper-noun lists.""" 

385 clean_tickers = [t for t in tickers if t not in _NOISE_TICKER_SYMBOLS] 

386 clean_proper = [w for w in proper if w not in _COMMON_WORDS] 

387 return clean_tickers, clean_proper

390def _dedup_entities(entities: list) -> list: 

391 """Deduplicate entity list while preserving insertion order (case-insensitive).""" 

392 seen: set = set() 

393 deduped = [] 

394 for e in entities: 

395 if e.lower() not in seen: 

396 seen.add(e.lower()) 

397 deduped.append(e) 

398 return deduped

401def _extract_title_entities(title: str) -> list: 

402 """Extract meaningful entities (names, tickers, numbers) from title.""" 

403 tickers, values, proper, kr_entities = _extract_raw_patterns(title) 

404 tickers, proper = _filter_entities(tickers, proper) 

405 

406 entities: list = [] 

407 entities.extend(values) 

408 entities.extend(tickers[:2]) 

409 entities.extend(proper[:2]) 

410 entities.extend(kr_entities[:3]) 

411 return _dedup_entities(entities)

414def _analyze_title_content(title: str) -> str: 

415 """Generate an analytical Korean description from a news title. 

416 

417 Instead of generic templates, this produces a title-derived summary 

418 that explains what the article is about and why it matters. 

419 """ 

420 title_lower = title.lower() 

421 title_stripped = title.strip() 

422 

423 # --- Korean title: extract meaning and add context --- 

424 kr_chars = len(re.findall(r"[가-힣]", title_stripped)) 

425 if kr_chars > len(title_stripped) * 0.3: 

426 return _analyze_korean_title(title_stripped) 

427 

428 # --- English title: translate key concepts and add analysis --- 

429 return _analyze_english_title(title_stripped, title_lower)

472# --------------------------------------------------------------------------- 

473# Output hygiene for synthesised descriptions (2026-08-06). 

474# 

475# The 2026-08-06 backfill dry-run synthesised 417 replacements and a large share 

476# read worse than the title restatement they replaced. Three causes, each fixed 

477# below and pinned in tests/test_enrichment_synthetic_quality.py. 

478# --------------------------------------------------------------------------- 

479 

480# Outlet suffix: `… - The Hill`, `… | 공감신문`. Bounded and digit-free so an 

481# informative subtitle ("삼성전자 - 2분기 영업이익 14조 원") is left alone; the 

482# previous single-token pattern (`\S+$`) stripped 프리진경제 but not "The Hill". 

483# 

484# **Whitespace before the delimiter is required.** With `\s*` (zero allowed) the 

485# hyphen inside a compound matched: "…net inflows to a multi-month high." lost 

486# its tail to "…a multi", and "미국-이란 평화 협정을 지적했습니다." became "…미국". 

487# A false positive here destroys a sentence, while a false negative only leaves 

488# an outlet name behind — so the rule errs toward keeping text. Cost: an outlet 

489# written without a leading space ("…지지-[美증시 특징주]") is no longer stripped. 

490_SOURCE_SUFFIX_RE = re.compile(r"\s+[-–—|]\s*(?![^-–—|]*\d)[^-–—|]{2,20}$")

619def _strip_particle(token: str) -> str: 

620 """Remove one trailing grammatical particle, longest match first.""" 

621 for particle in _KO_PARTICLES: 

622 if token.endswith(particle) and len(token) - len(particle) >= 2: 

623 return token[: -len(particle)] 

624 return token

627def korean_keywords(title: str, limit: int = 3) -> list: 

628 """Topical Hangul keywords from a title, or ``[]`` when none survive. 

629 

630 Returning an empty list is a real outcome: a `주요 키워드:` label with one 

631 filler word under it is worse than no label, so callers omit the clause. 

632 """ 

633 seen: list = [] 

634 for raw in re.findall(r"[가-힣]{2,}", title): 

635 token = _strip_particle(raw) 

636 if len(token) < 2 or token in seen: 

637 continue 

638 if token in _KO_STOPWORDS or token in _KO_UNIT_WORDS: 

639 continue 

640 if token.endswith(_KO_PREDICATE_ENDINGS): 

641 continue 

642 seen.append(token) 

643 if len(seen) >= limit: 

644 break 

645 return seen if len(seen) >= 2 else []

648def _join_clauses(head: str, *tails: str) -> str: 

649 """Join a title with generated clauses without doubling terminators.""" 

650 text = head.strip() 

651 if text and not text.endswith(_SENTENCE_ENDERS): 

652 text += "." 

653 for tail in tails: 

654 tail = tail.strip() 

655 if tail: 

656 text = f"{text} {tail}".strip() 

657 return text

660def _analyze_korean_title(title: str) -> str: 

661 """Analyze a Korean news title and generate a fact-based description. 

662 

663 Produces clean, factual descriptions based on the title content 

664 rather than speculative or template-heavy commentary. 

665 """ 

666 clean = _strip_source_suffix(title) 

667 pct = re.search(r"(\d+(?:\.\d+)?)\s*%", title) 

668 amount = re.search(r"(\d[\d,.]*)\s*(?:억|조|만|원|달러)", title) 

669 kr_entities = korean_keywords(title) 

670 

671 # Build context suffix from extracted data 

672 context_parts = [] 

673 if pct: 

674 context_parts.append(f"{pct.group(1)}% 변동") 

675 if amount: 

676 context_parts.append(f"{amount.group(0)}") 

677 context_suffix = f" ({', '.join(context_parts)})" if context_parts else "" 

678 

679 # Circuit breaker / market crash 

680 if any(kw in title for kw in ["서킷브레이커", "사이드카"]): 

681 trigger = "매수 사이드카" if "매수" in title else "서킷브레이커/사이드카" 

682 return _join_clauses(clean[:120], f"{trigger} 발동{context_suffix}.") 

683 

684 # Crash / panic 

685 if any(kw in title for kw in ["폭락", "급락", "패닉"]): 

686 return _join_clauses(clean[:120], f"{context_suffix.strip()} 급락 관련 보도.".strip()) 

687 

688 # Semiconductor 

689 if any(kw in title for kw in ["삼성전자", "하이닉스", "반도체"]): 

690 return _join_clauses(clean[:120], f"{context_suffix.strip()} 반도체 섹터 보도.".strip()) 

691 

692 # Fixed-response categories via dict table 

693 for kw_set, category_label in _KOREAN_TITLE_KW_SETS: 

694 if any(kw in title for kw in kw_set): 

695 return _join_clauses(clean[:120], f"{context_suffix.strip()} {category_label}".strip()) 

696 

697 # Crypto majors 

698 if any(kw in title for kw in ["비트코인", "이더리움", "알트코인", "리플"]): 

699 return _join_clauses(clean[:120], f"{context_suffix.strip()} 암호화폐 시장 보도.".strip()) 

700 

701 # DeFi / digital assets 

702 if any(kw in title for kw in ["디파이", "디지털자산", "가상자산", "코인", "블록체인"]): 

703 return _join_clauses(clean[:120], f"{context_suffix.strip()} 디지털 자산 보도.".strip()) 

704 

705 # AI / tech 

706 if any(kw in title for kw in ["AI", "인공지능", "챗봇", "생성형"]): 

707 return _join_clauses(clean[:120], f"{context_suffix.strip()} AI 산업 보도.".strip()) 

708 

709 # EV / battery 

710 if any(kw in title for kw in ["2차전지", "배터리", "전기차", "EV"]): 

711 return _join_clauses(clean[:120], f"{context_suffix.strip()} 전기차·배터리 보도.".strip()) 

712 

713 # Foreign / institutional flow 

714 if any(kw in title for kw in ["외국인", "기관", "순매수", "순매도", "수급"]): 

715 return _join_clauses(clean[:120], f"{context_suffix.strip()} 수급 동향 보도.".strip()) 

716 

717 # Earnings 

718 if any(kw in title for kw in ["실적", "어닝", "매출", "영업이익"]): 

719 return _join_clauses(clean[:120], f"{context_suffix.strip()} 기업 실적 보도.".strip()) 

720 

721 # Fallback: use clean title + entity context 

722 # A figure lifted from the title informs; a bag of the title's own words 

723 # does not. When both are available the number wins and the keyword tail is 

724 # dropped — emitting "(3.2% 변동) 주요 키워드: 국내, 관광객, 늘어난." padded 

725 # the sentence without adding anything. 

726 if context_suffix: 

727 return _join_clauses(clean[:120], context_suffix.strip()) 

728 if kr_entities: 

729 entity_str = ", ".join(kr_entities) 

730 return _join_clauses(clean[:120], f"주요 키워드: {entity_str}.") 

731 return _join_clauses(clean[:150]) if len(clean) > 15 else _join_clauses(title)

761def _analyze_english_title(title: str, title_lower: str) -> str: 

762 """Analyze an English news title and generate a specific Korean description.""" 

763 _subj = _extract_title_entities(title) 

764 # Use only named entities (not raw numbers/prices) for the subject prefix 

765 _named = [e for e in _subj if not re.match(r"^[\d$,.%+\-]+$", e)] 

766 

767 # Extract numbers/percentages/prices from title for specificity 

768 pct_match = re.search(r"(\d[\d,.]*)\s*%", title) 

769 # Full-word units allow a leading space ("$80 thousand"); single-letter 

770 # units must be immediately adjacent ("$73K"). Word units are matched first 

771 # so the "t" in "thousand" is never mistaken for a trillion suffix, and the 

772 # letter alt has no leading \s* so "$73 Kelvin" stays "$73" (not "$73K"). 

773 # No trailing \b: "$5MM" still degrades to "$5M" instead of dropping detail. 

774 price_match = re.search( 

775 r"\$(\d[\d,.]*(?:\.\d+)?)" 

776 r"(?:\s*(trillion|billion|million|thousand)s?|([KMBT]))?", 

777 title, 

778 re.IGNORECASE, 

779 ) 

780 points_match = re.search(r"(\d[\d,.]*)\s*(?:points?|pts?)", title, re.IGNORECASE) 

781 

782 # Clean title: remove trailing source names for use as base description 

783 clean_title = re.sub( 

784 r"\s*[-–—|]\s*(?:Reuters|Bloomberg|CNBC|AP|MarketWatch|CoinDesk|Cointelegraph|TradingView|Yahoo Finance|The Motley Fool|Investopedia)\s*$", 

785 "", 

786 title, 

787 flags=re.IGNORECASE, 

788 ).strip() 

789 # Also remove generic trailing patterns (require a space before separator to avoid 

790 # matching hyphenated words like "All-Time") 

791 clean_title = re.sub(r"\s+[–—|]\s*[A-Z][\w\s]{2,20}$", "", clean_title).strip() 

792 

793 # Build detail string from extracted numbers 

794 detail_parts = [] 

795 if pct_match: 

796 detail_parts.append(f"{pct_match.group(1)}% 변동") 

797 if price_match: 

798 # Normalize the magnitude suffix to a canonical single letter so the 

799 # extracted detail never contradicts the full number in the body text 

800 # (e.g. "$73K" — not a misleading "$73" — for a "$73,000" article). 

801 _word_unit = (price_match.group(2) or "").lower() 

802 _letter_unit = (price_match.group(3) or "").upper() 

803 unit = {"trillion": "T", "billion": "B", "million": "M", "thousand": "K"}.get(_word_unit, _letter_unit) 

804 detail_parts.append(f"${price_match.group(1)}{unit}") 

805 if points_match: 

806 detail_parts.append(f"{points_match.group(1)}포인트") 

807 detail_str = ", ".join(detail_parts) 

808 

809 # Determine event category and add specific context 

810 

811 # Build extra context from extracted numbers 

812 extra = f" ({detail_str})" if detail_str else "" 

813 

814 # --- Dynamic categories: inject detail_str into result --- 

815 if any(kw in title_lower for kw in ["crash", "plunge", "tumble", "sink", "slump", "dive"]): 

816 return f"{clean_title[:120]}.{extra} 급락 관련 보도." 

817 

818 if any(kw in title_lower for kw in ["rally", "surge", "soar", "jump", "climb", "gain"]): 

819 return f"{clean_title[:120]}.{extra} 상승 관련 보도." 

820 

821 if any(kw in title_lower for kw in ["drop", "fall ", "decline", "down ", "lower", "slip"]): 

822 return f"{clean_title[:120]}.{extra}" 

823 

824 if any(kw in title_lower for kw in ["rise", "up ", "higher", "advance"]): 

825 return f"{clean_title[:120]}.{extra}" 

826 

827 if any(kw in title_lower for kw in ["oil", "crude", "brent", "wti"]): 

828 return f"{clean_title[:120]}.{extra} 유가 관련 보도." 

829 

830 if any(kw in title_lower for kw in ["s&p", "nasdaq", "dow", "futures", "stock market"]): 

831 return f"{clean_title[:120]}.{extra}" 

832 

833 # --- Fixed-suffix categories via dict table --- 

834 for kw_set, suffix in _ENGLISH_TITLE_KW_SETS: 

835 if any(kw in title_lower for kw in kw_set): 

836 return f"{clean_title[:120]}.{extra}" if not suffix else f"{clean_title[:120]}.{extra} {suffix}" 

837 

838 # Default: use cleaned title with entity context 

839 _meaningful = [e for e in _named if len(e) > 2] 

840 if _meaningful: 

841 subj = ", ".join(_meaningful[:2]) 

842 return f"{clean_title[:120]}.{extra} {subj} 관련 보도." 

843 if detail_str: 

844 return f"{clean_title[:120]}.{extra}" 

845 return f"{clean_title[:140]}." if len(clean_title) > 15 else title

848def generate_synthetic_description( 

849 title: str, 

850 source: str, 

851 context_map: Optional[Dict[str, str]] = None, 

852) -> str: 

853 """Generate a contextual description when RSS provides none. 

854 

855 Uses title content analysis to produce meaningful, specific summaries 

856 instead of generic template-based descriptions. 

857 """ 

858 # Try title-based analysis first (produces specific, analytical content) 

859 analysis = _analyze_title_content(title) 

860 if analysis and analysis != title and len(analysis) > 20: 

861 return analysis 

862 

863 # Fallback: build a title-derived description with source context 

864 label = _get_source_label(source, context_map or {}) 

865 entities = _extract_title_entities(title) 

866 entity_str = ", ".join(entities[:3]) if entities else "" 

867 

868 # Exchange-specific fallback 

869 if any(kw in source.lower() for kw in ["binance", "bybit", "okx", "upbit"]): 

870 return f"{label} 공지. " + (f"{entity_str} 관련." if entity_str else f"{title[:80]}.") 

871 

872 # Build a title-condensed description instead of boilerplate 

873 # Clean trailing source names from the title 

874 # `\s+` (not `\s*`): a hyphen inside a compound is not a delimiter. With zero 

875 # spaces allowed this cut "(BTC-USD:Cryptocurrency)" down to "(BTC" — 21 such 

876 # titles in the corpus. Cost of the stricter rule is an outlet glued on 

877 # without a space, which is the cheaper failure. 

878 clean_title = re.sub(r"\s+[-–—|]\s*\S+$", "", title).strip() 

879 # For Korean titles, extract a condensed version 

880 kr_chars = len(re.findall(r"[가-힣]", clean_title)) 

881 if kr_chars > len(clean_title) * 0.3: 

882 core = clean_title[:80] if len(clean_title) > 80 else clean_title 

883 if entity_str: 

884 return f"{core}. {entity_str} 관련 보도." 

885 return f"{core}." 

886 

887 # English: use cleaned title with source context 

888 if label and label != source: 

889 return f"{label} 보도. {clean_title[:100]}" 

890 return clean_title[:150] if len(clean_title) > 15 else title

893def _extract_overlap_keywords(title: str, text: str) -> tuple[set, set]: 

894 """Extract normalized keyword sets from title and text for relevance scoring.""" 

895 title_tokens = set(re.findall(r"[A-Za-z0-9$%]{3,}|[가-힣]{2,}", title.lower())) 

896 text_tokens = set(re.findall(r"[A-Za-z0-9$%]{3,}|[가-힣]{2,}", text.lower())) 

897 return title_tokens, text_tokens

900def _is_title_related_description(title: str, desc: str) -> bool: 

901 """Return True if description appears related to title content.""" 

902 if not title or not desc: 

903 return True 

904 # Backward-compatible: do not aggressively reject when title itself is weak. 

905 if len(title.strip()) < 18: 

906 return True 

907 if _is_desc_duplicate_of_title(desc, title): 

908 return True 

909 

910 title_tokens, desc_tokens = _extract_overlap_keywords(title, desc) 

911 if not title_tokens or not desc_tokens: 

912 return True 

913 

914 if title_tokens & desc_tokens: 

915 return True 

916 

917 title_entities = {e.lower() for e in _extract_title_entities(title)} 

918 if len(title_entities) < 2 and len(title_tokens) < 3: 

919 return True 

920 if title_entities and not (title_entities & desc_tokens): 

921 return False 

922 return True

925def _is_desc_duplicate_of_title(desc: str, title: str) -> bool: 

926 """Return True if desc is essentially a duplicate of title (no extra info). 

927 

928 Checks: 

929 - Normalized text equality (lowercase, stripped punctuation/whitespace) 

930 - desc contains >= 80% of title characters 

931 - desc length < 1.3x title length (insufficient additional info) 

932 - Word-token Jaccard similarity > 0.7 

933 """ 

934 if not desc or not title: 

935 return False 

936 

937 # Normalize: lowercase, remove punctuation and whitespace 

938 norm_desc = _NORM_RE.sub("", desc.lower()) 

939 norm_title = _NORM_RE.sub("", title.lower()) 

940 

941 # Exact normalized match 

942 if norm_desc == norm_title: 

943 return True 

944 

945 # Sequence-based similarity: catches near-duplicates without false positives 

946 # from bag-of-characters approach 

947 if norm_title and len(norm_title) > 5: 

948 ratio = SequenceMatcher(None, norm_desc, norm_title).ratio() 

949 if ratio > 0.8 and len(norm_desc) < len(norm_title) * 1.3: 

950 return True 

951 

952 # Word-token Jaccard similarity (skip for very short titles to avoid noise) 

953 desc_tokens = set(desc.lower().split()) 

954 title_tokens = set(title.lower().split()) 

955 union = desc_tokens | title_tokens 

956 if len(union) >= 4: 

957 intersection = desc_tokens & title_tokens 

958 if len(intersection) / len(union) > 0.7: 

959 return True 

960 

961 return False
