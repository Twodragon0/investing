80def extract_entities(text: str) -> dict[str, list[str]]: 

81 """Extract named entities from text. 

82 

83 Returns dict with keys: crypto, stock, index, person, org, theme 

84 Each value is a list of canonical entity names found. 

85 """ 

86 results: dict[str, list[str]] = { 

87 "crypto": [], 

88 "stock": [], 

89 "index": [], 

90 "person": [], 

91 "org": [], 

92 "theme": [], 

93 } 

94 

95 text_upper = text.upper() 

96 text_lower = text.lower() 

97 

98 for canonical, aliases in _CRYPTO_ENTITIES.items(): 

99 for alias in aliases: 

100 if alias in text or alias.upper() in text_upper: 

101 if canonical not in results["crypto"]: 

102 results["crypto"].append(canonical) 

103 break 

104 

105 for canonical, aliases in _STOCK_ENTITIES.items(): 

106 for alias in aliases: 

107 if alias in text or alias.upper() in text_upper: 

108 if canonical not in results["stock"]: 

109 results["stock"].append(canonical) 

110 break 

111 

112 for canonical, aliases in _INDEX_ENTITIES.items(): 

113 for alias in aliases: 

114 if alias in text: 

115 if canonical not in results["index"]: 

116 results["index"].append(canonical) 

117 break 

118 

119 for canonical, aliases in _PERSON_ENTITIES.items(): 

120 for alias in aliases: 

121 if alias in text: 

122 if canonical not in results["person"]: 

123 results["person"].append(canonical) 

124 break 

125 

126 for canonical, aliases in _ORG_ENTITIES.items(): 

127 for alias in aliases: 

128 if alias in text: 

129 if canonical not in results["org"]: 

130 results["org"].append(canonical) 

131 break 

132 

133 for theme, keywords in _THEME_KEYWORDS.items(): 

134 for kw in keywords: 

135 if kw.lower() in text_lower: 

136 if theme not in results["theme"]: 

137 results["theme"].append(theme) 

138 break 

139 

140 return results

143def group_related_items(items: list[dict[str, Any]], title_key: str = "title") -> dict[str, list[dict[str, Any]]]: 

144 """Group news items by shared entities. 

145 

146 Returns dict mapping group labels to lists of related items. 

147 Items sharing entities are grouped together. 

148 """ 

149 # Extract entities for each item 

150 item_entities: list[tuple[dict, dict]] = [] 

151 for item in items: 

152 title = item.get(title_key, "") 

153 desc = item.get("description", "") 

154 text = f"{title} {desc}" 

155 entities = extract_entities(text) 

156 item.setdefault("_entities", entities) 

157 item_entities.append((item, entities)) 

158 

159 # Build entity → items index 

160 entity_to_items: dict[str, list[int]] = defaultdict(list) 

161 for idx, (_, entities) in enumerate(item_entities): 

162 for category, names in entities.items(): 

163 for name in names: 

164 key = f"{category}:{name}" 

165 entity_to_items[key].append(idx) 

166 

167 # Find groups (entities shared by 2+ items) 

168 groups: dict[str, list[dict]] = {} 

169 used_indices: set[int] = set() 

170 

171 # Sort by group size (largest first) 

172 sorted_entities = sorted(entity_to_items.items(), key=lambda x: len(x[1]), reverse=True) 

173 

174 for entity_key, indices in sorted_entities: 

175 if len(indices) < 2: 

176 continue 

177 # Filter out already-grouped items 

178 new_indices = [i for i in indices if i not in used_indices] 

179 if len(new_indices) < 2: 

180 continue 

181 

182 category, name = entity_key.split(":", 1) 

183 label = _format_group_label(category, name) 

184 group_items = [item_entities[i][0] for i in new_indices] 

185 groups[label] = group_items 

186 used_indices.update(new_indices) 

187 

188 # Add ungrouped items 

189 ungrouped = [item_entities[i][0] for i in range(len(items)) if i not in used_indices] 

190 if ungrouped: 

191 groups["기타 뉴스"] = ungrouped 

192 

193 return groups

196def extract_market_signals(items: list[dict[str, Any]], title_key: str = "title") -> dict[str, Any]: 

197 """Extract market signal summary from a collection of news items. 

198 

199 Returns a dict with signal counts and dominant themes. 

200 """ 

201 all_entities: dict[str, list[str]] = defaultdict(list) 

202 theme_counts: dict[str, int] = defaultdict(int) 

203 

204 for item in items: 

205 title = item.get(title_key, "") 

206 desc = item.get("description", "") 

207 text = f"{title} {desc}" 

208 entities = extract_entities(text) 

209 

210 for category, names in entities.items(): 

211 all_entities[category].extend(names) 

212 for theme in entities.get("theme", []): 

213 theme_counts[theme] += 1 

214 

215 # Count entity frequencies 

216 entity_freq: dict[str, dict[str, int]] = {} 

217 for category, names in all_entities.items(): 

218 freq: dict[str, int] = defaultdict(int) 

219 for name in names: 

220 freq[name] += 1 

221 entity_freq[category] = dict(sorted(freq.items(), key=lambda x: x[1], reverse=True)) 

222 

223 # Determine dominant themes 

224 sorted_themes = sorted(theme_counts.items(), key=lambda x: x[1], reverse=True) 

225 

226 return { 

227 "entity_frequencies": entity_freq, 

228 "dominant_themes": sorted_themes[:5], 

229 "total_items": len(items), 

230 "theme_coverage": len(theme_counts), 

231 }
