1#!/usr/bin/env python3 

2r"""Dependabot 커밋 메시지에서 `update-type`(semver 변화 등급)을 판정한다. 

3 

4## 왜 이 모듈이 필요한가 

5 

6`dependabot-auto-merge.yml` 은 `dependabot/fetch-metadata` 액션의 `update-type` 

7출력으로 자동 머지 대상을 가른다. 그런데 **그 액션은 `pull_request` / 

8`pull_request_target` 전용**이다 (2026-09-16 upstream 확인): 

9 

10* `src/dependabot/verified_commits.ts:20` 이 이벤트 페이로드의 `pull_request` 키를 

11 요구하고, 없으면 *"Make sure you're triggering this action on the `pull_request` 

12 or `pull_request_target` events"* 로 실패한다. 

13* `action.yml` 의 inputs 5개(`alert-lookup`, `compat-lookup`, `github-token`, 

14 `skip-commit-verification`, `skip-verification`) 중 **PR 번호를 넘기는 것이 없다.** 

15 

16따라서 `schedule` 스위퍼(열린 dependabot PR 을 주기적으로 순회해 머지하는 구조)는 

17그 액션을 쓸 수 없다. 이 모듈은 그 자리를 메운다. 

18 

19## 취약하지 않은 이유 — 같은 출처를 읽는다 

20 

21PR 제목을 파싱하는 방식이 아니다. `fetch-metadata` **자신이 읽는 원본**이 dependabot 

22커밋 메시지이고(`src/dependabot/update_metadata.ts:69-81`), 같은 데이터를 

23`gh api repos/{o}/{r}/pulls/{n}/commits` 로 얻을 수 있다. 

24 

25## upstream 동작을 그대로 재현한다 (아티팩트 포함) 

26 

27판정 결과가 액션과 **달라지면 안 된다.** 스위퍼가 액션보다 넓게 머지하는 순간 

28"자동 머지 대상 집합"이 조용히 바뀌기 때문이다. 그래서 알려진 파싱 아티팩트까지 

29재현한다: 

30 

31 chore(deps): update boto3 requirement from <2,>=1.43.92 to >=1.43.93,<2 

32 

33이 줄에서 `from` 캡처는 `2,>=1.43.92` 가 된다 — 정규식의 `\S*?` 가 이전 제약의 

34상한 `<2,` 중 `<` 만 삼키고 멈추기 때문이다. 결과적으로 `lastParts[0] == "2,>=1"`, 

35`nextParts[0] == "1"` 이 되어 **패치 크기 bump 가 `semver-major` 로 떨어진다**. 

36#1321 · #1329 가 그렇게 분류돼 자동 머지 경로를 타지 못했다(실측). 

37 

38이것은 의미 판단이 아니라 버그로 보이지만, **여기서 고치지 않는다.** 고치면 스위퍼가 

39액션과 다르게 동작한다. 분류를 바로잡는 것은 별도 결정이며, 그때는 이 모듈이 아니라 

40호출부의 정책으로 다뤄야 한다(그래야 "액션과 일치" 라는 이 모듈의 계약이 유지된다). 

41 

42고정 대상 upstream: `dependabot/fetch-metadata` v3.1.0 

43(`25dd0e34f4fe68f24cc83900b1fe3fe149efef98`). 로직 출처는 

44`src/dependabot/update_metadata.ts:69-176`, `src/dependabot/output.ts:10-14,79-86`. 

45upstream 을 올릴 때 `tests/fixtures/dependabot_commits/manifest.json` 의 기대값을 

46다시 확인할 것. 

47 

48## 사용법 

49 

50 python3 scripts/tools/dependabot_update_type.py --branch <브랜치명> [커밋메시지파일] 

51 gh api repos/o/r/pulls/123/commits --jq '.[0].commit.message' \ 

52 | python3 scripts/tools/dependabot_update_type.py --branch dependabot/pip/x 

53 

54판정 불가(형식 불일치, dependabot 브랜치 아님 등)는 빈 문자열을 출력하고 종료코드 1. 

55액션도 메타데이터를 못 찾으면 실패 코드를 낸다. 

56"""

60import argparse 

61import re 

62import sys

66#: upstream `output.ts:10-14`. 여러 의존성이 한 PR 에 있으면 **가장 큰** 변화를 쓴다. 

67UPDATE_TYPES_PRIORITY = ( 

68 "version-update:semver-major", 

69 "version-update:semver-minor", 

70 "version-update:semver-patch", 

71) 

72 

73#: upstream `update_metadata.ts:70-74`. 표기를 바꾸지 말 것 — 아티팩트까지 동일해야 한다. 

74_UPDATE_RE = re.compile(r"\b[Uu]pdate .* requirement from \S*? ?(?P<from>v?\d\S*) to \S*? ?(?P<to>v?\d\S*)") 

75_BUMP_RE = re.compile(r"^Bumps .* from (?P<from>v?\d[^ ]*) to (?P<to>v?\d[^ ]*)\.$", re.M) 

76_YAML_RE = re.compile(r"^-{3}\n(?P<dependencies>[\S\s]*?)\n^\.{3}\n", re.M)

79def calculate_update_type(last_version: str, next_version: str) -> str: 

80 """upstream `update_metadata.ts:159-176` 의 축자 이식. 

81 

82 semver 파서를 쓰지 않는다 — upstream 이 문자열을 `.` 으로 쪼개 비교할 뿐이라, 

83 파서를 쓰면 위 아티팩트가 재현되지 않는다. 

84 """ 

85 if not last_version or not next_version or last_version == next_version: 

86 return "" 

87 

88 last_parts = last_version.replace("v", "").split(".") 

89 next_parts = next_version.replace("v", "").split(".") 

90 

91 if last_parts[0] != next_parts[0]: 

92 return "version-update:semver-major" 

93 if len(last_parts) < 2 or len(next_parts) < 2 or last_parts[1] != next_parts[1]: 

94 return "version-update:semver-minor" 

95 return "version-update:semver-patch" 

96

98def parse_update_type(commit_message: str, branch_name: str) -> str: 

99 """커밋 메시지 + 브랜치명에서 `update-type` 을 판정한다. 판정 불가면 빈 문자열. 

100 

101 upstream 과 같은 조건을 요구한다 — YAML 조각이 있고 브랜치가 `dependabot` 으로 

102 시작할 것(`update_metadata.ts:80`). 둘 중 하나라도 어긋나면 액션도 메타데이터를 

103 만들지 않는다. 

104 """ 

105 yaml_match = _YAML_RE.search(commit_message) 

106 if not yaml_match or not branch_name.startswith("dependabot"): 

107 return "" 

108 

109 try: 

110 data = yaml.safe_load(yaml_match.group("dependencies")) 

111 except yaml.YAMLError: 

112 return "" 

113 if not isinstance(data, dict) or not data.get("updated-dependencies"): 

114 return "" 

115 

116 # upstream `update_metadata.ts:71-72`: `Bumps …` 는 메시지 **전체**에서(그 줄은 

117 # 보통 3번째다), `Update … requirement` 는 **첫 줄**에서만 찾는다. 순서도 같다 — 

118 # bump 가 우선이다. 

119 # 

120 # 한 변수에 담아 두는 이유: `(bump or update).group(...) if (bump or update)` 로 

121 # 쓰면 두 번 평가라 타입 narrowing 이 되지 않는다(basedpyright 

122 # `reportOptionalMemberAccess`, 2026-09-16 CI 에서 실제로 걸렸다). 

123 version_match = _BUMP_RE.search(commit_message) or _UPDATE_RE.search(commit_message.split("\n")[0]) 

124 prev = version_match.group("from") if version_match else "" 

125 nxt = version_match.group("to") if version_match else "" 

126 

127 levels: set[str] = set() 

128 for index, dependency in enumerate(data["updated-dependencies"]): 

129 if not isinstance(dependency, dict): 

130 continue 

131 # upstream `update_metadata.ts:100-103`: 첫 의존성만 커밋 메시지의 from/to 를 

132 # 물려받는다. 둘째부터는 YAML 의 값에만 의존한다. 

133 last_version = prev if index == 0 else "" 

134 next_version = dependency.get("dependency-version") or (nxt if index == 0 else "") 

135 levels.add(dependency.get("update-type") or calculate_update_type(last_version, next_version)) 

136 

137 for level in UPDATE_TYPES_PRIORITY: 

138 if level in levels: 

139 return level 

140 return ""

148
